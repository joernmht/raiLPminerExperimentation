%@ meta id=rail_reschedule family=milp schema=0.1.0
%@ name :: Rolling‑horizon railway timetable rescheduling and rolling‑stock reassignment
%@ desc :: MILP model integrating departure‑time adjustments, cancellations, short‑turning, stop‑skipping, additional service insertion and train‑to‑trip assignment under uncertain disruption durations.
%@ index I ordered=0 cyclic=0 :: Set of trips (original and potential additional services).
%@ index S ordered=0 cyclic=0 :: Set of available trains.
%@ param d0 shape=I kind=vector domain=time_duration :: Planned departure time of trip i.
%@ param dur shape=I kind=vector domain=time_duration :: Planned travel duration of trip i.
%@ param delta_short shape=- kind=scalar domain=time_duration :: Time saved by short‑turning a trip.
%@ param delta_skip shape=- kind=scalar domain=time_duration :: Time saved by stop‑skipping a trip.
%@ param tau shape=- kind=scalar domain=time_duration :: Minimum turnaround time between consecutive trips of the same train.
%@ param M shape=- kind=scalar domain=penalty_bigM :: Large constant for big‑M constraints.
%@ param c_delay shape=- kind=scalar domain=cost_weight :: Unit cost of delay.
%@ param c_cancel shape=- kind=scalar domain=cost_weight :: Penalty cost for cancelling a trip.
%@ param c_short shape=- kind=scalar domain=cost_weight :: Cost for applying short‑turn to a trip.
%@ param c_skip shape=- kind=scalar domain=cost_weight :: Cost for applying stop‑skip to a trip.
%@ param c_add shape=- kind=scalar domain=cost_weight :: Cost for inserting an additional service.
%@ param c_assign shape=- kind=scalar domain=cost_weight :: Cost per train‑to‑trip assignment.
%@ var d shape=I domain=continuous role=primary drole=- lo=0 hi=- :: Actual departure time of trip i.
%@ var a shape=I domain=continuous role=primary drole=- lo=0 hi=- :: Arrival time of trip i.
%@ var delay shape=I domain=non_negative role=primary drole=- lo=0 hi=- :: Delay of trip i.
%@ var y shape=I domain=binary role=primary drole=- lo=- hi=- :: Cancellation indicator (1 if trip i is cancelled).
%@ var z shape=I domain=binary role=primary drole=- lo=- hi=- :: Short‑turn indicator (1 if applied to trip i).
%@ var w shape=I domain=binary role=primary drole=- lo=- hi=- :: Stop‑skip indicator (1 if applied to trip i).
%@ var add shape=I domain=binary role=primary drole=- lo=- hi=- :: Additional service indicator (1 if an extra service is inserted for i).
%@ var x shape=I,S domain=binary role=primary drole=- lo=- hi=- :: Assignment of train s to trip i.
%@ obj sense=min name=total_cost combination=sum :: Minimize total operational cost (delay, penalties, and assignment).
%@ con delay_def kind=linear domain=- indicator=- :: Definition of delay.
%@ con arrival_lower kind=linear domain=- indicator=- :: Lower bound for arrival time (including short‑turn / stop‑skip).
%@ con arrival_upper kind=linear domain=- indicator=- :: Upper bound for arrival time (enforcing equality).
%@ con assignment_balance_low kind=linear domain=- indicator=- :: Assignment balance (lower side).
%@ con assignment_balance_up kind=linear domain=- indicator=- :: Assignment balance (upper side).
%@ con train_feasibility kind=linear domain=- indicator=- :: Time feasibility for consecutive trips on the same train.
\begin{align}
  \min\quad & \sum_{i \in \mathcal{I}} \mathit{c\_delay} \cdot delay_{i}
            + \sum_{i \in \mathcal{I}} \mathit{c\_cancel} \cdot y_{i}
            + \sum_{i \in \mathcal{I}} \mathit{c\_short} \cdot z_{i}
            + \sum_{i \in \mathcal{I}} \mathit{c\_skip} \cdot w_{i}
            + \sum_{i \in \mathcal{I}} \mathit{c\_add} \cdot add_{i}
            + \sum_{i \in \mathcal{I}, s \in \mathcal{S}} \mathit{c\_assign} \cdot x_{i,s}
            \tag{total\_cost} \\
  & delay_{i} - d_{i} + d0_{i} \ge 0 \qquad \forall i \in \mathcal{I} \tag{delay\_def} \\
  & a_{i} - d_{i} - dur_{i} + \mathit{delta\_short} \cdot z_{i} + \mathit{delta\_skip} \cdot w_{i} \ge 0 \qquad \forall i \in \mathcal{I} \tag{arrival\_lower} \\
  & a_{i} - d_{i} - dur_{i} + \mathit{delta\_short} \cdot z_{i} + \mathit{delta\_skip} \cdot w_{i} \le 0 \qquad \forall i \in \mathcal{I} \tag{arrival\_upper} \\
  & \sum_{s \in \mathcal{S}} x_{i,s} - 1 + y_{i} - add_{i} \ge 0 \qquad \forall i \in \mathcal{I} \tag{assignment\_balance\_low} \\
  & \sum_{s \in \mathcal{S}} x_{i,s} - 1 + y_{i} - add_{i} \le 0 \qquad \forall i \in \mathcal{I} \tag{assignment\_balance\_up} \\
  & d_{j} - a_{i} - \mathit{tau} + \mathit{M} - \mathit{M} \cdot x_{i,s} - \mathit{M} \cdot x_{j,s} \ge 0 \qquad \forall i \in \mathcal{I},\; \forall j \in \mathcal{I},\; i \neq j,\; \forall s \in \mathcal{S} \tag{train\_feasibility} \\
\end{align}
