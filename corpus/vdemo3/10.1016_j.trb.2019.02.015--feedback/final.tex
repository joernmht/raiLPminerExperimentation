%@ meta id=rail_reschedule family=milp schema=0.1.0
%@ name :: Railway timetable rescheduling with flexible stopping, short‑turning, retiming, reordering and cancellation
%@ desc :: MILP model that integrates flexible stopping (skip/add stops), flexible short‑turning (choice of turn stations), retiming (assignment to discrete time slots), implicit reordering (slot exclusivity) and train cancellation while respecting platform capacity. Objective minimizes a proxy for passenger delay (stop penalty) plus cancellation penalties.
%@ index T ordered=0 cyclic=0 :: Set of trains.
%@ index S ordered=0 cyclic=0 :: Set of stations.
%@ index H ordered=0 cyclic=0 :: Set of discrete time slots.
%@ param p_stop shape=- kind=scalar domain=- :: Penalty per train stop (proxy for passenger delay).
%@ param cancel_penalty shape=- kind=scalar domain=- :: Penalty for cancelling a train.
%@ param cap shape=S kind=scalar domain=- :: Platform capacity at each station.
%@ param M shape=- kind=scalar domain=- :: Large constant for big‑M constraints.
%@ param st_allowed shape=T,S kind=matrix domain=- :: Indicator if station s is allowed as a short‑turn point for train t.
%@ param after shape=T,S,S kind=matrix domain=- :: Indicator if station s' is downstream of station s on train t’s route.
%@ var x shape=T,S,H domain=binary role=primary drole=- lo=- hi=- :: Occupancy of train t at station s during time slot h (stop decision and timing).
%@ var z shape=T,S domain=binary role=auxiliary drole=- lo=- hi=- :: Short‑turn decision: 1 if train t terminates at station s.
%@ var c shape=T domain=binary role=auxiliary drole=- lo=- hi=- :: Cancellation decision for train t.
%@ obj sense=min name=total_cost combination=sum :: Minimize weighted passenger delay (stop penalty) and cancellation penalties.
%@ con cancel_disable kind=- domain=- indicator=- :: Disable occupancy for cancelled trains.
%@ con platform_capacity kind=- domain=- indicator=- :: Platform capacity per station and time slot.
%@ con train_single_slot kind=- domain=- indicator=- :: At most one station per train per time slot.
%@ con shortturn_allowed kind=- domain=- indicator=- :: Short‑turn only at allowed stations.
%@ con shortturn_single kind=- domain=- indicator=- :: At most one short‑turn per train.
%@ con shortturn_after kind=- domain=- indicator=- :: No occupancy after a short‑turn point (big‑M).
\begin{align}
  \min\quad & p\_stop \cdot \sum_{t \in T, s \in S, h \in H} x_{t,s,h} + cancel\_penalty \cdot \sum_{t \in T} c_{t} \tag{total\_cost} \\
  & x_{t,s,h} \le 1 - c_{t} \qquad \forall t \in T, s \in S, h \in H \tag{cancel\_disable} \\
  & \sum_{t \in T} x_{t,s,h} \le cap_{s} \qquad \forall s \in S, h \in H \tag{platform\_capacity} \\
  & \sum_{s \in S} x_{t,s,h} \le 1 - c_{t} \qquad \forall t \in T, h \in H \tag{train\_single\_slot} \\
  & z_{t,s} \le st\_allowed_{t,s} \qquad \forall t \in T, s \in S \tag{shortturn\_allowed} \\
  & \sum_{s \in S} z_{t,s} \le 1 \qquad \forall t \in T \tag{shortturn\_single} \\
  & x_{t,s',h} \le M \cdot (2 - z_{t,s} - after_{t,s,s'}) \qquad \forall t \in T, s \in S, s' \in S, h \in H \tag{shortturn\_after} \\
\end{align}
