%@ meta id=rail_passenger_reschedule family=milp schema=0.1.0
%@ name :: Passenger-oriented railway timetable rescheduling
%@ desc :: MILP integrating train schedule adjustments (delay, cancellation) and passenger reassignment to minimize generalized travel time (in‑vehicle time, waiting time and transfer penalty) while respecting train capacities and maximum allowed extra delay.
%@ index T ordered=0 cyclic=0 :: Set of trains.
%@ index S ordered=0 cyclic=0 :: Set of stations.
%@ index P ordered=0 cyclic=0 :: Set of passengers.
%@ param origin shape=P kind=vector domain=station :: Origin station of passenger p.
%@ param dest shape=P kind=vector domain=station :: Destination station of passenger p.
%@ param travel shape=T,S,S kind=scalar domain=time :: In‑vehicle travel time of train t from station i to station j.
%@ param wait shape=P,T,S,S kind=scalar domain=time :: Waiting time for passenger p to board train t on arc (i,j).
%@ param cap shape=T kind=scalar domain=passenger :: Capacity of train t.
%@ param M_transfer shape=- kind=scalar domain=penalty :: Penalty cost per passenger transfer.
%@ param Dmax shape=- kind=scalar domain=time :: Maximum allowed extra delay for any train.
%@ var f shape=P,T,S,S domain=binary role=primary drole=- lo=- hi=- :: Passenger p uses train t on arc (i,j).
%@ var c shape=T domain=binary role=auxiliary drole=- lo=- hi=- :: Train t runs (1) or is cancelled (0).
%@ var delta shape=T domain=non_negative role=auxiliary drole=- lo=- hi=- :: Extra delay of train t.
%@ obj sense=min name=gen_travel_time combination=sum :: Minimize total generalized travel time (in‑vehicle, waiting and transfer penalties).
%@ con origin_assign kind=- domain=- indicator=- :: Each passenger departs from its origin on exactly one arc.
%@ con dest_assign kind=- domain=- indicator=- :: Each passenger arrives at its destination on exactly one arc.
%@ con flow_intermediate kind=- domain=- indicator=- :: Flow conservation for intermediate stations.
%@ con capacity kind=- domain=- indicator=- :: Train capacity limits.
%@ con run_link kind=- domain=- indicator=- :: Passengers can use only running trains.
%@ con delay_limit kind=- domain=- indicator=- :: Extra delay bounded by Dmax.
\begin{align}
  \min\quad & \sum_{p \in P, t \in T, i \in S, j \in S} travel_{t,i,j} \cdot f_{p,t,i,j} + \sum_{p \in P, t \in T, i \in S, j \in S} wait_{p,t,i,j} \cdot f_{p,t,i,j} + M\_transfer \cdot \sum_{p \in P, t \in T, i \in S, j \in S} f_{p,t,i,j} \tag{gen_travel_time} \\
  & \sum_{t \in T, j \in S} f_{p,t,origin_{p},j} = 1 \qquad \forall p \in P \tag{origin_assign} \\
  & \sum_{t \in T, i \in S} f_{p,t,i,dest_{p}} = 1 \qquad \forall p \in P \tag{dest_assign} \\
  & \sum_{t \in T, j \in S} f_{p,t,i,j} - \sum_{t \in T, j \in S} f_{p,t,j,i} = 0 \qquad \forall p \in P, \forall i \in S \tag{flow_intermediate} \\
  & \sum_{p \in P, i \in S, j \in S} f_{p,t,i,j} \le cap_{t} \cdot c_{t} \qquad \forall t \in T \tag{capacity} \\
  & f_{p,t,i,j} \le c_{t} \qquad \forall p \in P, \forall t \in T, \forall i \in S, \forall j \in S \tag{run_link} \\
  & delta_{t} \le Dmax \qquad \forall t \in T \tag{delay_limit}
\end{align}
