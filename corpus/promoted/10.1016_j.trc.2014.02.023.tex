\begin{align}
  \min\quad & \left\{f_{e} , f_{t} , f_{d}\right\} \tag{eq\_0001} \\
  & f_{t} = \sum_{i = 1}^{N} \left(A_{i , a} - D_{i , a}\right) / N \tag{eq\_0002} \\
  & f_{e} = \sum_{i = 1}^{N} \sum_{x = 0}^{L_{i}} u_{i , x} \cdot F_{i , x} \left(v\right) w_{i} \tag{eq\_0003\_a} \\
  & f_{e} = \sum_{i = 1}^{N} \sum_{t = 0}^{A_{i , a}} u_{i , t} \cdot F_{i , t} \left(v\right) v_{i} \left(t\right) w_{i} \tag{eq\_0003\_b} \\
  & f_{e} = \sum_{i = 1}^{N} \sum_{t = 0}^{A_{i , a}} u_{i , t} \cdot P_{i , t} \cdot w_{i} \tag{eq\_0003\_c} \\
  & f_{n} = \sum_{i = 1}^{N} \sum_{k = 1}^{M} \text{sgn} \left(A_{i , k , a} - A_{i , k , o}\right) \tag{eq\_0004} \\
  & D_{i , a , k} \geq D_{i , o , k} \tag{eq\_0005} \\
  & D_{i , a , k} - A_{i , a , k} \geq B_{i , k}^{min} \tag{eq\_0006} \\
  & V_{i} \left(t\right) \leq V_{i ,\text{max}} \tag{eq\_0007} \\
  & X_{i} \left(A_{i , a}\right) = L_{i} \tag{eq\_0010} \\
  & X_{i} \left(t\right) - X_{j} \left(t\right) \geq max \left\{\left[v_{j} \left(t\right)^{2} - v_{i} \left(t\right)^{2}\right] / 2 \cdot a + d_{min} , d_{min}\right\} \tag{eq\_0011} \\
  & F = \mathit{alpha} \times f_{t} + \mathit{beta} \times f_{e} + \mathit{gamma} \times f_{d} \tag{eq\_0015} \\
  & f = \mathit{alpha} \times \left|f_{t} - \bar{f}_{t}\right| / \bar{f}_{t} + \mathit{beta} \times \left|f_{e} - \bar{f}_{e}\right| / \bar{f}_{e} + \mathit{gamma} \times \left|f_{d} - \bar{f}_{d}\right| / \bar{f}_{d} \tag{eq\_0016} \\
\end{align}
