  & \mathit{tau}_{i}^{C} \cdot x_{k,i}^{C} \leq t_{k,i}^{dC} - t_{k,i}^{aC} \qquad \forall i \in N_{C}, k \in T_{n} \tag{eq\_0012\_a} \\
  & t_{k,i}^{dC} - t_{k,i}^{aC} \leq M \cdot x_{k,i}^{C} \qquad \forall i \in N_{C}, k \in T_{n} \tag{eq\_0012\_b} \\
  & x_{k,i}^{C} \leq 1 - \mathit{rho}_{k,j} \qquad \forall j \in N_{m}, i \in N_{C}, i \in IC_{j} \tag{eq\_0013} \\
  & \left|\right. t_{l , i}^{d H} - t_{k , i}^{d H} \left|\right. \geq h_{i}^{d H} - M \cdot \left(2 - \mathit{rho}_{l , j} - \mathit{rho}_{k , j}\right) , \tag{eq\_0014} \\
  & t_{l,i}^{dH} - t_{k,i}^{dH} \geq h_{i}^{dH} - M \cdot (3 - y_{k,l,i} - \mathit{rho}_{l,j} - \mathit{rho}_{k,j}) \tag{eq\_0015} \\
  & t_{k , i}^{d H} - t_{l , i}^{d H} \geq h_{i}^{d H} - M \cdot \left(3 - y_{l , k , i} - \mathit{rho}_{l , j} - \mathit{rho}_{k , j}\right) . \tag{eq\_0016} \\

  & t_{k , i}^{a H} - t_{l , i}^{d H} \geq h^{d a H} - M \cdot \left(5 - \mathit{rho}_{k , j} - \mathit{rho}_{l , j} - y_{l , k , i} - \mathit{lambda}_{k , i , q}^{H} - \mathit{lambda}_{l , i , q}^{H}\right) , \text{for} \text{all} q \in P_{i} , \tag{eq\_0029} \\
