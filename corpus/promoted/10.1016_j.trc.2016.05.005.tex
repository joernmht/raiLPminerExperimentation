  & y_{\mathit{ij}} \geq T - T_{\text{max}} \cdot \left(1 - z_{\mathit{ij}}\right) , \forall \left(i , j\right) \in \mathcal{A}_{\text{run}} \cup \mathcal{A}_{\text{dwell}} \cup \mathcal{A}_{\text{safe}} \cup \mathcal{A}_{\text{regular}} , \tag{eq\_0010} \\
  & y_{\mathit{ij}} \geq 0 , \forall \left(i , j\right) \in \mathcal{A}_{\text{run}} \cup \mathcal{A}_{\text{dwell}} \cup \mathcal{A}_{\text{safe}} \cup \mathcal{A}_{\text{regular}} . \tag{eq\_0011} \\

  & \frac{T}{f_{\mathit{ij}}} - \frac{K}{f_{\mathit{ij}}} \leq x_{\mathit{ij}} \leq \frac{T}{f_{\mathit{ij}}} + \frac{K}{f_{\mathit{ij}}} , \forall \left(i , j\right) \in \mathcal{A}_{\text{regular}} . \tag{eq\_0014} \\

  & \begin{aligned} & z_{\mathit{ij}} + z_{i^{'} j^{'}} + z_{\mathit{ii}^{'}} + z_{\mathit{jj}^{'}} = 2 \cdot \left(w_{\mathit{ii}^{'} \mathit{jj}^{'}} + v_{\mathit{ii}^{'} \mathit{jj}^{'}}\right) \\ & \forall \left(i , j\right) , \left(i^{'} , j^{'}\right) \in \mathcal{A}_{\text{run}} , \left(i , i^{'}\right) , \left(j , j^{'}\right) \in \mathcal{A}_{\text{safe}} \cup \mathcal{A}_{\text{regular}} .\end{aligned} \tag{eq\_0022} \\
