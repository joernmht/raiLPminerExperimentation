  & d_{C_n,t} = \sum_{Kappa \in \mathcal{K}} d_{Kappa,C_n,t} \tag{eq\_0004} \\

  & q_{ab,C_n} = \sum_{I_n \in C_n, m_n \in \mathcal{M}} q_{ab,m_n,I_n} \tag{eq\_0012\_a} \\

  & t_{Arr,m_n,I_n} = t_{Dep,m_n,I_1} + \sum_{i = 1}^{n} t_{I_i,I_iplus1,m_n} + \sum_{i = 1}^{n - 1} t_{dwell,m_n,I_i} \tag{eq\_0014} \\
  & t_{Dep,m_n,I_n} = t_{Arr,m_n,I_n} + t_{Hold,m_n,I_n} \tag{eq\_0015} \\
  & t_{b u s}^{I_{n} I_{n ^{'}}} = \frac{\mathit{Delta}_{u}^{I_{n} I_{n ^{'}}}}{v_{b u s} \left(1 - q_{a b}^{m_{n} I_{n ^{'}}} / q_{a b}^{I_{n}} \right)} \tag{eq\_0017\_a} \\
