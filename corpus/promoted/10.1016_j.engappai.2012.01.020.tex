  & \left(\mathit{pi}_{j} - \mathit{pi}_{i} - l_{a}\right) \mathit{modT} \leq u_{a} - l_{a} , \forall a = \left(i , j\right) \in A , \tag{eq\_0001} \\
  & l_{a} \leq \mathit{pi}_{j} - \mathit{pi}_{i} - T \cdot z_{a} \leq u_{a} \tag{eq\_0002} \\
  \min\quad & \sum_{i \in R} w_{i} \left(\mathit{arr}_{i d_{i}} - \mathit{arr}_{i o_{i}}\right) \tag{eq\_0003} \\
  & t_{i b '} + s_{i b '} \leq \mathit{arr}_{i b '} - \mathit{arr}_{i b} \leq T - 1 , \forall i \in R , \forall b , b ' \in \left[o_{i} , d_{i}\right] \tag{eq\_0004} \\
  & t_{i b} \leq \mathit{arr}_{i b} - \mathit{arr}_{j b} - z_{i j b} \cdot T \leq T - t_{j b} , \forall i , j \in R , \forall b \in \left[o_{i} , d_{i}\right] \cap \left[o_{j} , d_{j}\right] \tag{eq\_0005} \\
  & l_{i} \leq \mathit{arr}_{i o_{i}} - t_{i o_{i}} \leq u_{i} , \forall i \in R , \tag{eq\_0006} \\
\end{align}
