  \min\quad & \sum_{c \in C} \sum_{l \in L_{c}} \left(\mathit{gamma}_{l}^{\text{Fix}} \cdot u_{l} + \mathit{gamma}^{\text{Pen}} \cdot s_{l} + \underset{a \in A^{\text{Light}} \cap A_{l}}{\sum} \mathit{gamma}_{l a}^{\text{Light}} \cdot x_{l a} + \underset{a \in A^{\text{Train}} \cap A_{l}}{\sum} \mathit{gamma}_{l a}^{\text{Active}} \cdot x_{l a} + \underset{a \in A^{\text{Train}} \cap A_{l}}{\sum} \mathit{gamma}_{l a}^{\text{Passive}} \cdot y_{l a}\right) + \underset{a \in A^{\text{Bust}}}{\sum} \mathit{gamma}^{\text{Bust}} \cdot w_{a} \tag{eq\_0001} \\
  & \text{s}.\text{t}. \underset{a \in A_{i}^{\text{In}} \cap A_{c}}{\sum} \left(x_{c a} + y_{c a}\right) = \underset{a \in A_{i}^{\text{Out}} \cap A_{c}}{\sum} \left(x_{c a} + y_{c a}\right) \forall i \in N , \forall c \in C , \tag{eq\_0002} \\

  & \sum_{c \in C^{P u s h} : a \in A_{c}} x_{c a} \geq 1 \forall a \in A^{\text{Push}} , \tag{eq\_0005} \\

  & x_{c a} + y_{c a} \leq r_{c a} \cdot K \forall a \in A^{\text{Train}} , c \in C : a \in A_{c} , \tag{eq\_0007} \\
  & r_{c a} + r_{\hat{c} a} \leq 1 \forall c \in C , \hat{c} \in \left(\hat{C}\right)_{c} , a \in A^{\text{Train}} \cap A_{c} \cap A_{\hat{c}} , \tag{eq\_0008} \\
  & u_{l} = \sum_{c \in C : l \in L_{c}} \underset{a \in A^{\text{Sink}} \cap A_{c}}{\sum} y_{c a} \forall l \in L , \tag{eq\_0009} \\
  & u_{l} - s_{l} \leq B_{l} \forall l \in L , \tag{eq\_0010} \\
