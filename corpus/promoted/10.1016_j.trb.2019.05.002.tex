\begin{align}
  & a_{i j} = \mathit{pi}_{j} - \mathit{pi}_{i} + z_{i j} \cdot T \tag{eq\_0001} \\
  \min\quad & \underset{\left(i , j\right) \in \mathbf{\mathcal{A}}_{run} \cup \mathbf{\mathcal{A}}_{dwell}}{\sum} \mathit{alpha}_{i j} \cdot \left(\mathit{pi}_{j} - \mathit{pi}_{i} + z_{i j} \cdot T\right) \tag{eq\_0002} \\
  & l_{i j} \leq \mathit{pi}_{j} - \mathit{pi}_{i} + z_{i j} \cdot T \leq u_{i j} \forall \left(i , j\right) \in \mathbf{\mathcal{A}} \tag{eq\_0003} \\
  & 0 \leq \mathit{pi}_{i} < T \forall i \in \mathbf{\mathcal{E}} \tag{eq\_0004} \\
  & z_{i j} \in \left\{0 , 1\right\} \forall \left(i , j\right) \in \mathbf{\mathcal{A}} \tag{eq\_0005} \\
  & z_{i j} + z_{ip jp} + z_{i ip} + z_{j jp} = 2 \cdot c_{i ip j jp} \forall \left(i , j\right) , \left(ip , jp\right) \in \mathbf{\mathcal{A}}_{run} , \left(i , ip\right) , \left(j , jp\right) \in \mathbf{\mathcal{A}}_{head} \tag{eq\_0006} \\
  & 0 \leq c_{i ip j jp} \leq 2 \forall \left(i , j\right) , \left(ip , jp\right) \in \mathbf{\mathcal{A}}_{run} , \left(i , ip\right) , \left(j , jp\right) \in \mathbf{\mathcal{A}}_{head} \tag{eq\_0007} \\
  & c_{i ip j jp} \in \mathcal{N}_{0} \forall \left(i , j\right) , \left(ip , jp\right) \in \mathbf{\mathcal{A}}_{run} , \left(i , ip\right) , \left(j , jp\right) \in \mathbf{\mathcal{A}}_{head} \tag{eq\_0008} \\
  & \mathit{theta}_{i j} = \left|a_{i j} - \left[\frac{T}{f_{l}}\right]\right| \forall \left(i , j\right) \in \mathbf{\mathcal{A}}_{reg}^{l} , l \in \mathbf{\mathcal{L}} \tag{eq\_0009} \\
  & - \mathit{theta}_{i j} \leq a_{i j} - \left[\frac{T}{f_{l}}\right] \leq \mathit{theta}_{i j} \forall \left(i , j\right) \in \mathbf{\mathcal{A}}_{reg}^{l} , l \in \mathbf{\mathcal{L}} \tag{eq\_0011} \\
  & 0 \leq \mathit{theta}_{i j} \leq \mathit{beta}_{l} \forall \left(i , j\right) \in \mathbf{\mathcal{A}}_{reg}^{l} , l \in \mathbf{\mathcal{L}} \tag{eq\_0012} \\
  & h^{p} = \lceil \frac{T}{N_{t}} \rceil \tag{eq\_0013} \\
  & P_{m a x} = \mathit{phi} \cdot \left(h^{p} - l^{m i n}\right) \tag{eq\_0014} \\
  & h_{i j}^{p l_underline o w e r} = l_{i j} + \left(h^{p} - l^{m i n}\right) \forall \left(i , j\right) \in \mathbf{\mathcal{A}}_{head} \tag{eq\_0016} \\
  & h_{i j}^{p u_underline p p e r} = u_{i j} - \left(h^{p} - l^{m i n}\right) \forall \left(i , j\right) \in \mathbf{\mathcal{A}}_{head} \tag{eq\_0017} \\
  & \mathit{delta}_{i j} = w_{i j}^{'} \cdot \left(P_{m a x} - \mathit{phi} \cdot \left(a_{i j} - l_{i j}\right)\right) + w_{i j}^{'' '} \cdot \left(P_{m a x} - \mathit{phi} \cdot \left(u_{i j} - a_{i j}\right)\right) \forall \left(i , j\right) \in \mathbf{\mathcal{A}}_{head} \tag{eq\_0019} \\
  & w_{i j}^{'} + w_{i j}^{''} + w_{i j}^{'' '} = 1 \forall \left(i , j\right) \in \mathbf{\mathcal{A}}_{head} \tag{eq\_0020} \\
  & w_{i j}^{'} , w_{i j}^{''} , w_{i j}^{'' '} \in \left\{0 , 1\right\} \forall \left(i , j\right) \in \mathbf{\mathcal{A}}_{head} \tag{eq\_0021} \\
  & l_{i j} \leq a_{i j} \leq h_{i j}^{p l_underline o w e r} + M_{1} \cdot \left(1 - w_{i j}^{'}\right) \forall \left(i , j\right) \in \mathbf{\mathcal{A}}_{head} \tag{eq\_0022} \\
  & w_{i j}^{''} \cdot h_{i j}^{p l_underline o w e r} < a_{i j} \leq h_{i j}^{p u_underline p p e r} + M_{2} \cdot \left(1 - w_{i j}^{''}\right) \forall \left(i , j\right) \in \mathbf{\mathcal{A}}_{head} \tag{eq\_0023} \\
  & w_{i j}^{'' '} \cdot h_{i j}^{p u_underline p p e r} < a_{i j} \leq u_{i j} \forall \left(i , j\right) \in \mathbf{\mathcal{A}}_{head} \tag{eq\_0024} \\
  & h_{s_{k}} = \underset{\left(i , j\right) \in \mathbf{\mathcal{A}}_{head}^{s_{k}}}{max} \left\{l_{i j} , T - u_{i j}\right\} \tag{eq\_0025} \\
  & d_{j jp} = u_{j jp} - l_{j jp} \tag{eq\_0026} \\
  & z_{i j} + z_{ip jp} + z_{i ip} + z_{j jp} = 2 \cdot o_{i ip j jp} + p_{i ip j jp} \forall \left(i , ip\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}} , \left(j , jp\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}} \cup \mathbf{\mathcal{A}}_{thru}^{s_{k}} , \left(i , j\right) , \left(ip , jp\right) \in \mathbf{\mathcal{A}}_{head}^{s_{k}} , s_{k} \in \mathbf{\mathcal{S}} \tag{eq\_0028} \\
  & y_{i ip j jp} + y_{j jp i ip} = p_{i ip j jp} \forall \left(i , ip\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}} , \left(j , jp\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}} \cup \mathbf{\mathcal{A}}_{thru}^{s_{k}} , \left(i , j\right) , \left(ip , jp\right) \in \mathbf{\mathcal{A}}_{head}^{s_{k}} , s_{k} \in \mathbf{\mathcal{S}} \tag{eq\_0029} \\
  & o_{i ip j jp} , p_{i ip j jp} , y_{i ip j jp} , y_{j jp i ip} \in \left\{0 , 1\right\} \forall \left(i , ip\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}} , \left(j , jp\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}} \cup \mathbf{\mathcal{A}}_{thru}^{s_{k}} , \left(i , j\right) , \left(ip , jp\right) \in \mathbf{\mathcal{A}}_{head}^{s_{k}} , s_{k} \in \mathbf{\mathcal{S}} \tag{eq\_0030} \\
  & \begin{matrix} & & - y_{j jp i ip} \cdot d_{j jp} - \left(1 - p_{i ip j jp}\right) \cdot \left(u_{j jp} - l_{i ip}\right) \leq a_{i ip} - a_{j jp} \leq y_{i ip j jp} \cdot d_{i ip} + \left(1 - p_{i ip j jp}\right) \cdot \left(u_{i ip} - l_{j jp}\right) \\ & & \forall \left(i , ip\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}} , \left(j , jp\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}} \cup \mathbf{\mathcal{A}}_{thru}^{s_{k}} , s_{k} \in \mathbf{\mathcal{S}}\end{matrix} \tag{eq\_0031} \\
  & x_{i ip} \in \left\{0 , 1\right\} \forall \left(i , ip\right) \in \mathbf{\mathcal{A}}_{dwell} \tag{eq\_0032} \\
  & \frac{1}{M} \underset{\left(j , jp\right) \in \mathbf{\mathcal{A}}_{thru}^{s_{k}}}{\sum} y_{i ip j jp} \leq x_{i ip} \leq \underset{\left(j , jp\right) \in \mathbf{\mathcal{A}}_{thru}^{s_{k}}}{\sum} y_{i ip j jp} \forall \left(i , ip\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}} , s_{k} \in \mathbf{\mathcal{S}} \tag{eq\_0033} \\
  & a_{i ip} \leq l_{i ip} + 2 \cdot h_{s_{k}} \cdot \left(\underset{\left(j , jp\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}}}{\sum} y_{i ip j jp}\right) + h_{s_{k}} \cdot \left(\underset{\left(j , jp\right) \in \mathbf{\mathcal{A}}_{thru}^{s_{k}}}{\sum} y_{i ip j jp}\right) + x_{i ip} \cdot \left(h_{s_{k}} - l_{i ip}\right) \forall \left(i , ip\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}} , s_{k} \in \mathbf{\mathcal{S}} \tag{eq\_0034} \\
  & \underset{\left(i , ip\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}}}{\sum} p_{i ip j jp} \leq O_{s_{k}}^{m a x} \forall \left(j , jp\right) \in \mathbf{\mathcal{A}}_{dwell}^{s_{k}} \cup \mathbf{\mathcal{A}}_{thru}^{s_{k}} , s_{k} \in \mathbf{\mathcal{S}} \tag{eq\_0035} \\
\end{align}
