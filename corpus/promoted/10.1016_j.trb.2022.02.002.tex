  \min\quad & \sum_{f \in F} \sum_{g \in A_{f}} c_{g}^{f} \cdot x_{g}^{f} \tag{eq\_0001} \\

  & x_{v}^{f} = \sum_{g \in \mathit{delta}_{f}^{-} \left(v\right) \backslash A_{f}^{e}} x_{g}^{f} \forall v \in V_{f}^{d} \cup V_{f}^{s e} , \forall f \in F \tag{eq\_0011} \\
  & x_{v}^{f} = \sum_{g \in \mathit{delta}_{f}^{+} \left(v\right) \backslash A_{f}^{o}} x_{g}^{f} \forall v \in V_{f}^{a} , \forall f \in F \tag{eq\_0013} \\
