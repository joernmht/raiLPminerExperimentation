\begin{align}
  & C \left(\mathit{pi}_{1} , 1\right) = t \left(\mathit{pi}_{1} , 1\right) \tag{eq\_0001} \\
  & C \left(\mathit{pi}_{i} , 1\right) = C \left(\mathit{pi}_{i - 1} , 1\right) + t \left(\mathit{pi}_{i} , 1\right) i = 2 , \ldots , n \tag{eq\_0002} \\

  C\quad & \left(\mathit{pi}_{i} , j\right) = max C \left(\mathit{pi}_{i - 1} , j\right) , C \left(\mathit{pi}_{i} , j - 1\right) \left.\right\} + t \left(\mathit{pi}_{i} , j\right) i = 2 , \ldots , n ; \tag{eq\_0004} \\
  & j = 2, \ldots, m \tag{eq\_0005\_a} \\
  & \text{Makespan}, f_{C_{max}} = C(\mathit{pi}_n,m) \tag{eq\_0005\_b} \\
  & \text{Total \cdot tardiness}, f_{\mathit{total} \mathit{tardiness}} = \sum_{i = 1}^{n} max \left[0 , L_{i}\right] \tag{eq\_0006} \\
  & f_{\text{total idle time}} = \sum_{j=2}^{m} \left( C(\mathit{pi}_1,j-1) + \sum_{i=2}^{n} \max\{ C(\mathit{pi}_i,j-1) - C(\mathit{pi}_{i-1},j), 0 \} \right) \tag{eq\_0007} \\
