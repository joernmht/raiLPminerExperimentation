  & d_{i,j} = a_{i,j} + s \cdot x_{i,j} \qquad \forall i \in I, j \in J \tag{eq\_0001} \\
  & a_{i,\left(j + 1\right)} = d_{i,j} + \mathit{tau}_{j,\left(j + 1\right)} \qquad \forall i \in I, j \in J \setminus \{ J_{last} \} \tag{eq\_0002} \\
  & hmin \leq d_{\left(i + 1\right),j} - d_{i,j} \qquad \forall i \in I \setminus \{ I_{last} \}, j \in J \tag{eq\_0003\_a} \\
  & d_{\left(i + 1\right),j} - d_{i,j} \leq hmax \qquad \forall i \in I \setminus \{ I_{last} \}, j \in J \tag{eq\_0003\_b} \\
  & \sum_{i \in I} y_{k,i} = 1 \qquad \forall k \in K \tag{eq\_0004} \\
  & y_{k,i} \leq x_{i,o_{k}} \qquad \forall k \in K, i \in I \tag{eq\_0005} \\
  & y_{k,i} \leq x_{i,e_{k}} \qquad \forall k \in K, i \in I \tag{eq\_0006} \\
  & M \cdot z_{k,i} + t_{o_{k},k} \geq d_{i,o_{k}} \qquad \forall k \in K, i \in I \tag{eq\_0007} \\
  & M \cdot (z_{k,i} - 1) + t_{o_{k},k} \leq d_{i,o_{k}} \qquad \forall k \in K, i \in I \tag{eq\_0008} \\
  & y_{k,i} \leq z_{k,i} \qquad \forall k \in K, i \in I \tag{eq\_0009} \\
  & M \cdot \sum_{p \in I} y_{k,p} \geq M \cdot (x_{i,j} - 1) + c_{i,j} - (g_{k} - 1) + M \cdot (z_{k,i} - 1) \qquad \forall i \in I, j \in J, k \in Khat_{j} \tag{eq\_0011} \\
  & c_{i,1} = C - \sum_{k \in Khat_1} g \cdot y_{k,i} \qquad \forall i \in I \tag{eq\_0012} \\
  & c_{i,j} = c_{i,j-1} + \sum_{k \in K_j} g \cdot y_{k,i} - \sum_{k \in Khat_j} g \cdot y_{k,i} \qquad \forall i \in I, j \in J \setminus \{1\} \tag{eq\_0013} \\
  & 0 \leq c_{i,j} \qquad \forall i \in I, j \in J \tag{eq\_0014\_a} \\
  & c_{i,j} \leq C \qquad \forall i \in I, j \in J \tag{eq\_0014\_b} \\
  \min\quad & \mathit{eta} \tag{eq\_0017} \\
  & w_{k} \geq M \cdot (y_{k,i} - 1) + d_{i,o_{k}} - t_{o_{k},k} \qquad \forall i \in I, k \in K \tag{eq\_0018} \\
  & w_{k} \geq 0 \qquad \forall k \in K \tag{eq\_0019} \\
  & \mathit{eta} \geq w_{k} \qquad \forall k \in K \tag{eq\_0020} \\
  & x_{i j} \in \left\{0 , 1\right\} , \forall i \in I , j \in J \tag{eq\_0021} \\
  & y_{k i} \in \left\{0 , 1\right\} , \forall k \in K , i \in I \tag{eq\_0022} \\
  & z_{k i} \in \left\{0 , 1\right\} , \forall k \in K , i \in I \tag{eq\_0023} \\
  & d_{1 j} \in \left[A , A + S_{1 j}\right] , \forall j \in J \tag{eq\_0024} \\
