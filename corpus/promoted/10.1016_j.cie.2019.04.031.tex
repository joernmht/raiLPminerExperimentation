  & h_{\mathit{min}} \leq d_{j}^{1} - d_{j - 1}^{1} \leq h_{\mathit{max}} \forall j , j - 1 \in {N} . \tag{eq\_0004} \\
  & h_{\mathit{min}} \leq d_{j}^{\left|S\right| + 1} - d_{j - 1}^{\left|S\right| + 1} \leq h_{\mathit{max}} \forall j , j - 1 \in \underset{̲}{N} . \tag{eq\_0005} \\
  & a_{j}^{u} = d_{j}^{u - 1} + r^{u - 1} , \forall u , u - 1 \in {S} , \forall j \in {N} \text{or} \forall u , u - 1 \in \underset{̲}{S} , \forall j \in \underset{̲}{N} . \tag{eq\_0006} \\
  & d_{j}^{u} = a_{j}^{u} + s^{u} \forall u \in {S} , \forall j \in {N} \text{or} \forall u \in \underset{̲}{S} , \forall j \in \underset{̲}{N} . \tag{eq\_0007} \\
  & Q_{j}^{u} = \sum_{u ' = 1}^{u} \sum_{v = u + 1}^{\left|S\right|} \sum_{t = 1}^{d_{j}^{u '}} P^{u ' , v} \left(t\right) , 1 \leq u < \left|S\right| , j = 1 . \tag{eq\_0008} \\
  & Q_{j}^{u} = \sum_{u ' = 1}^{u} \sum_{v = u + 1}^{\left|S\right|} \sum_{t = d_{j - 1}^{u '} + 1}^{d_{j}^{u '}} P^{u ' , v} \left(t\right) , 1 \leq u < \left|S\right| , 1 < j \leq \left|{N}\right| . \tag{eq\_0009} \\
  & Q_{j}^{u} = \sum_{u ' = 1}^{u} \sum_{v = u + 1}^{\left|S\right|} \sum_{t \in T} P^{u ' , v} \left(t\right) \cdot \left[1 - z_{j}^{u '} \left(t\right)\right] 1 \leq u < \left|S\right| , j = 1 \tag{eq\_0010} \\
  & Q_{j}^{u} = \sum_{u ' = 1}^{u} \sum_{v = u + 1}^{\left|S\right|} \sum_{t \in T} P^{u ' , v} \left(t\right) \cdot \left[z_{j - 1}^{u '} \left(t\right) - z_{j}^{u '} \left(t\right)\right] 1 \leq u < \left|S\right| , 1 < j \leq \left|{N}\right| . \tag{eq\_0011} \\
  & Q_{j}^{u} = \sum_{u ' = \left|S\right| + 1}^{u} \sum_{v = u + 1}^{\left|2 S\right|} \sum_{t \in T} P^{u ' , v} \left(t\right) \cdot \left[1 - z_{j}^{u '} \left(t\right)\right] \left|S\right| + 1 \leq u < \left|2 \cdot S\right| , j = 1 . \tag{eq\_0012} \\
  & Q_{j}^{u} = \sum_{u ' = \left|S\right| + 1}^{u} \sum_{v = u + 1}^{\left|2 S\right|} \sum_{t \in T} P^{u ' , v} \left(t\right) \cdot \left[z_{j - 1}^{u '} \left(t\right) - z_{j}^{u '} \left(t\right)\right] \left|S\right| + 1 \leq u < \left|2 \cdot S\right| , 1 < j \leq \left|\underset{̲}{N}\right| . \tag{eq\_0013} \\
  & Q_{j}^{u} \leq \sum_{k \in K} y_{j , k}^{\mathit{up}} \cdot c_{k} \forall u \in {S} , \forall j \in {N} . \tag{eq\_0014} \\
  & Q_{j}^{u} \leq \sum_{k \in K} y_{j , k}^{\mathit{dn}} \cdot c_{k} \forall u \in \underset{̲}{S} , \forall j \in \underset{̲}{N} . \tag{eq\_0015} \\
  & d_{j}^{u} = \sum_{t = 1}^{\left|T\right|} \left(t - 1\right) \cdot \left[z_{j}^{u} \left(t\right) - z_{j}^{u} \left(t - 1\right)\right] \forall u \in {S} , \forall j \in {N} , t > 1 \text{or} \forall u \in \underset{̲}{S} , \forall j \in \underset{̲}{N} , t > 1 . \tag{eq\_0016} \\
  & z_{j}^{u} \left(t - 1\right) \leq z_{j}^{u} \left(t\right) \forall u \in {S} , \forall j \in {N} , t > 1 \text{or} \forall u \in \underset{̲}{S} , \forall j \in \underset{̲}{N} , t > 1 . \tag{eq\_0017} \\
  & 0 \leq {\mathit{CT}}_{k} - \sum_{j \in {N}} y_{j , k}^{\mathit{up}} \cdot z_{j}^{1} \left(t\right) + \sum_{j ' \in \underset{̲}{N}} y_{j ' , k}^{\mathit{dn}} \cdot z_{j '}^{\left|2 S\right|} \left(t - \mathit{sigma}\right) \forall k \in K , t > \mathit{sigma} . \tag{eq\_0018} \\
  & 0 \leq \underset{̲}{\mathit{CT}}_{k} - \sum_{j ' \in \underset{̲}{N}} y_{j ' , k}^{\mathit{dn}} \cdot z_{j '}^{\left|S\right| + 1} \left(t\right) + \sum_{j \in {N}} y_{j , k}^{\mathit{up}} \cdot z_{j}^{\left|S\right|} \left(t - \mathit{sigma}\right) \forall k \in K , t > \mathit{sigma} . \tag{eq\_0019} \\
  & \sum_{k \in K} y_{j , k}^{\mathit{up}} = 1 \forall j \in {N} . \tag{eq\_0020} \\
  & \sum_{k \in K} y_{j , k}^{\mathit{dn}} = 1 \forall j \in \underset{̲}{N} . \tag{eq\_0021} \\
  & e_{j}^{\mathit{up}} \leq e_{j - 1}^{\mathit{up}} , 1 < j \leq \left|{N}\right| , \tag{eq\_0022} \\
  & \sum_{j = 1}^{\left|{N}\right| - 1} d_{j}^{1} \cdot \left(e_{j}^{\mathit{up}} - e_{j + 1}^{\mathit{up}}\right) + d_{\left|{N}\right|}^{1} \cdot e_{\left|{N}\right|}^{\mathit{up}} \geq {T}^{\mathit{up}} , \tag{eq\_0023} \\
  & \sum_{j = 1}^{\left|{N}\right| - 2} d_{j}^{1} \cdot \left(e_{j + 1}^{\mathit{up}} - e_{j + 2}^{\mathit{up}}\right) + d_{\left|{N}\right| - 1}^{1} \cdot e_{\left|{N}\right|}^{\mathit{up}} < {T}^{\mathit{up}} . \tag{eq\_0024} \\
  & e_{j}^{\mathit{dn}} \leq e_{j - 1}^{\mathit{dn}} , 1 < j \leq \left|\underset{̲}{N}\right| , \tag{eq\_0025} \\
  & \sum_{j = 1}^{\left|\underset{̲}{N}\right| - 1} d_{j}^{\left|S\right| + 1} \cdot \left(e_{j}^{\mathit{dn}} - e_{j + 1}^{\mathit{dn}}\right) + d_{\left|\underset{̲}{N}\right|}^{\left|S\right| + 1} \cdot e_{\left|\underset{̲}{N}\right|}^{\mathit{dn}} \geq {T}^{\mathit{down}} , \tag{eq\_0026} \\
  & \sum_{j = 1}^{\left|\underset{̲}{N}\right| - 2} d_{j}^{\left|S\right| + 1} \cdot \left(e_{j + 1}^{\mathit{dn}} - e_{j + 2}^{\mathit{dn}}\right) + d_{\left|\underset{̲}{N}\right| - 1}^{\left|S\right| + 1} \cdot e_{\left|\underset{̲}{N}\right|}^{\mathit{dn}} < {T}^{\mathit{down}} . \tag{eq\_0027} \\
  & \begin{matrix}W = \sum_{j \in {N}} \sum_{u = 1}^{\left|S\right| - 1} \sum_{v = u + 1}^{\left|S\right|} \sum_{t \in T} P^{u , v} \left(t\right) \cdot \left[z_{j - 1}^{u} \left(t\right) - z_{j}^{u} \left(t\right)\right] \cdot \left(d_{j}^{u} - t + 1\right) + \\ \sum_{j \in \underset{̲}{N}} \sum_{u = \left|S\right| + 1}^{\left|2 S\right| - 1} \sum_{v = u + 1}^{\left|2 S\right|} \sum_{t \in T} P^{u , v} \left(t\right) \cdot \left[z_{j - 1}^{u} \left(t\right) - z_{j}^{u} \left(t\right)\right] \cdot \left(d_{j}^{u} - t + 1\right) .\end{matrix} \tag{eq\_0028} \\
  & \mathit{EA}_{j} \left(t\right) = \sum_{k \in K} \sum_{u = 1}^{\left|S\right| - 1} \frac{1}{2 t_{\mathit{ab}}^{u}} y_{j , k}^{\mathit{up}} \cdot \left(M_{k} + m \cdot Q_{j}^{u}\right) \cdot \left(\mathit{vs}^{u}\right)^{2} \cdot \left[z_{j}^{u} \left(t\right) - z_{j}^{u} \left(t - t_{\mathit{ab}}^{u}\right)\right] , \forall j \in {N} , t > t_{\mathit{ab}}^{1} . \tag{eq\_0029} \\
  & \mathit{EA}_{j} \left(t\right) = \sum_{k \in K} \sum_{u = \left|S\right| + 1}^{\left|2 S\right| - 1} \frac{1}{2 t_{\mathit{ab}}^{u}} y_{j , k}^{\mathit{dn}} \cdot \left(M_{k} + m \cdot Q_{j}^{u}\right) \cdot \left(\mathit{vs}^{u}\right)^{2} \cdot \left[z_{j}^{u} \left(t\right) - z_{j}^{u} \left(t - t_{\mathit{ab}}^{u}\right)\right] , \forall j \in \underset{̲}{N} , t > t_{\mathit{ab}}^{\left|S\right| + 1} . \tag{eq\_0030} \\
  & f^{u} = R_{0} + R_{1} \cdot \mathit{vs}^{u} + R_{2} \cdot \left(\mathit{vs}^{u}\right)^{2} , \forall u \in {S} \cup \underset{̲}{S} . \tag{eq\_0031} \\
  & \mathit{EB}_{j} \left(t\right) = \sum_{k \in K} \sum_{u = 1}^{\left|S\right| - 1} y_{j , k}^{\mathit{up}} \cdot \left(M_{k} + m \cdot Q_{j}^{u}\right) \cdot f^{u} \cdot \mathit{vs}^{u} \cdot \left[z_{j}^{u} \left(t - t_{\mathit{ab}}^{u}\right) - z_{j}^{u} \left(t - t_{\mathit{bc}}^{u}\right)\right] , \forall j \in {N} , t > t_{\mathit{bc}}^{1} . \tag{eq\_0032} \\
  & \mathit{EB}_{j} \left(t\right) = \sum_{k \in K} \sum_{u = \left|S\right| + 1}^{\left|2 S\right| - 1} y_{j , k}^{\mathit{dn}} \cdot \left(M_{k} + m \cdot Q_{j}^{u}\right) \cdot f^{u} \cdot \mathit{vs}^{u} \cdot \left[z_{j}^{u} \left(t - t_{\mathit{ab}}^{u}\right) - z_{j}^{u} \left(t - t_{\mathit{bc}}^{u}\right)\right] , \forall j \in \underset{̲}{N} , t > t_{\mathit{bc}}^{\left|S\right| + 1} . \tag{eq\_0033} \\
  & \mathit{EC}_{j} \left(t\right) = 0 , \forall j \in {S} \cup \underset{̲}{S} , t \in T . \tag{eq\_0034} \\
  & \mathit{ED}_{j} \left(t\right) = \sum_{k \in K} \sum_{u = 1}^{\left|S\right| - 1} \frac{\mathit{eta}}{2 \left(r^{u} - t_{c d}^{u}\right)} \cdot y_{j , k}^{\mathit{up}} \cdot \left(M_{k} + m \cdot Q_{j}^{u}\right) \cdot \left(\mathit{vt}^{u}\right)^{2} \cdot \left[z_{j}^{u} \left(t - t_{c d}^{u}\right) - z_{j}^{u} \left(t - r^{u}\right)\right] , \forall j \in {N} , t > r^{1} . \tag{eq\_0035} \\
  & \mathit{ED}_{j} \left(t\right) = \sum_{k \in K} \sum_{u = \left|S\right| + 1}^{\left|2 S\right| - 1} \frac{\mathit{eta}}{2 \left(r^{u} - t_{c d}^{u}\right)} \cdot y_{j , k}^{\mathit{dn}} \cdot \left(M_{k} + m \cdot Q_{j}^{u}\right) \cdot \left(\mathit{vt}^{u}\right)^{2} \cdot \left[z_{j}^{u} \left(t - t_{c d}^{u}\right) - z_{j}^{u} \left(t - r^{u}\right)\right] , \forall j \in \underset{̲}{N} , t > r^{\left|S\right| + 1} . \tag{eq\_0036} \\
  & \begin{matrix}\mathit{CR} = \sum_{j \in {N}} e_{j}^{\mathit{up}} \cdot \left(\sum_{t = t_{\mathit{ab}}^{1} + 1}^{\left|T\right|} \mathit{fe} \left(t\right) \cdot \mathit{EA}_{j} \left(t\right) + \sum_{t = t_{\mathit{bc}}^{1} + 1}^{\left|T\right|} \mathit{fe} \left(t\right) \cdot \mathit{EB}_{j} \left(t\right) - \sum_{t = r^{1} + 1}^{\left|T\right|} \mathit{fe} \left(t\right) \cdot \mathit{ED}_{j} \left(t\right)\right) \\ + \sum_{j \in {N}} e_{j}^{\mathit{dn}} \cdot \left(\sum_{t = t_{\mathit{ab}}^{\left|S\right| + 1} + 1}^{\left|T\right|} \mathit{fe} \left(t\right) \cdot \mathit{EA}_{j} \left(t\right) + \sum_{t = t_{\mathit{bc}}^{\left|S\right| + 1} + 1}^{\left|T\right|} \mathit{fe} \left(t\right) \cdot \mathit{EB}_{j} \left(t\right) - \sum_{t = r_{}^{\left|S\right| + 1} + 1}^{\left|T\right|} \mathit{fe} \left(t\right) \cdot \mathit{ED}_{j} \left(t\right)\right) .\end{matrix} \tag{eq\_0037} \\
  & \mathit{CB} = \sum_{k \in K} \left(\sum_{j \in {N}} e_{j}^{\mathit{up}} \cdot y_{j , k}^{\mathit{up}} \cdot \mathit{fb} \cdot \mathit{PD}_{k} + \sum_{j \in \underset{̲}{N}} e_{j}^{\mathit{dn}} \cdot y_{j , k}^{\mathit{dn}} \cdot \mathit{fb} \cdot \mathit{PD}_{k}\right) . \tag{eq\_0038} \\
  & Z = \mathit{lambda}_{1} \cdot W + \mathit{lambda}_{2} \cdot \left(\mathit{CR} + \mathit{CB}\right) \tag{eq\_0039} \\
  \min\quad & \mathit{lambda}_{1} \cdot W + \mathit{lambda}_{2} \cdot \left(\mathit{CR} + \mathit{CB}\right) \tag{eq\_0040} \\
\end{align}
