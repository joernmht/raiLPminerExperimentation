  \min\quad & \sum_{k \in K} \sum_{w \in W} p^{w} \cdot f_{w}^{k} = \sum_{k \in K} \sum_{w \in W} p^{w} \cdot t_{w , k , s_{k}}^{a} \tag{eq\_0002} \\

  & t_{w , k , i + 1}^{a} - t_{w , k , i}^{d} \leq x_{w , k , i} \cdot g_{k}^{a} + x_{w , k , i + 1} \cdot g_{k}^{d} + v_{k , i}^{u} \forall k \in K , i \in N \backslash \left\{s_{k}\right\} , w \in W \tag{eq\_0008} \\
  & t_{w , k , i}^{d} - t_{w , l , i}^{d} \geq h_{i}^{d} - M \cdot \left(1 - y_{w , l , k , i}\right) \forall k , l \in K , k \neq l , i \in N \backslash \left\{s_{k} , s_{l}\right\} , w \in W \tag{eq\_0010} \\
  & t_{w , k , i}^{a} - t_{w , l , i}^{a} \geq h_{i}^{a} - M \cdot \left(1 - y_{w , l , k , i - 1}\right) \forall k , l \in K , k \neq l , i \in N \backslash \left\{o_{k} , o_{l}\right\} , w \in W \tag{eq\_0011} \\
  & \sum_{q \in Q_{i}} \mathit{lambda}_{w , k , i , q} = 1 \forall k \in K , i \in N , w \in W \tag{eq\_0012} \\
  & \mathit{lambda}_{w , k , i , 1} = 1 - x_{w , k , i} \forall k \in K , i \in N , w \in W \tag{eq\_0013} \\
  & t_{w , k , i}^{a} - t_{w , l , i}^{d} \geq h_{i}^{d a} - M \cdot \left(3 - y_{w , l , k , i} - \mathit{lambda}_{w , k , i , q} - \mathit{lambda}_{w , l , i , q}\right) \forall k , l \in K , k \neq l , i \in N , w \in W \tag{eq\_0014} \\
  & t_{w , k , i}^{d} \geq m_{i}^{end} - \left(1 - \mathit{delta}_{w , k , i}\right) \cdot M \forall k \in K , i \in N_{m} , w \in W \tag{eq\_0015} \\
  & t_{w , k , i}^{d} \leq m_{w , i}^{\text{start}} + \mathit{delta}_{w , k , i} \cdot M \forall k \in K , i \in N_{m} , w \in W \tag{eq\_0017} \\
