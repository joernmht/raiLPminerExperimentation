  \max\quad & \sum_{\forall l \in L} \sum_{\forall l^{'} \in L} \sum_{k \in S \left(l\right) \cap S \left(l^{'}\right)} p_{l l^{'}}^{k} \cdot x_{l l^{'}}^{k} + \sum_{\forall l \in L} \sum_{\forall l^{'} \in L} \sum_{\forall i \in S \left(l\right)} \sum_{\forall j \in S \left(l^{'}\right)} \mathit{alpha}_{l l^{'}}^{i j} \cdot p_{l l^{'}}^{i j} \tag{eq\_0001} \\
  & p_{l l^{'}}^{k} = \sum_{i = 1}^{k - 1} \sum_{j = k + 1}^{z} p_{l l^{'}}^{i j} \cdot k \in S \left(l\right) \cap S \left(l^{'}\right) , \forall l , l^{'} \in L \tag{eq\_0002} \\
  & \mathit{Arr}_{k}^{l} = t_{l}^{R} + \sum_{s = 1}^{k} t_{\left(s - 1\right) s}^{l} + \sum_{s = 0}^{k - 1} t_{s}^{l} \forall k \in S \left(l\right) , \forall l \in L \tag{eq\_0003} \\
  & \mathit{Dep}_{k}^{l^{'}} = t_{l^{'}}^{R} + \sum_{s = 1}^{k} t_{\left(s - 1\right) s}^{l^{'}} + \sum_{s = 0}^{k} t_{s}^{l^{'}} \forall k \in S \left(l\right) , \forall l^{'} \in L \tag{eq\_0004} \\
  & T_{l}^{min} \leq t_{l}^{R} \leq T_{l}^{max} \forall l \in L \tag{eq\_0005} \\
  & \mathit{Dep}_{min}^{l} \leq \mathit{Dep}_{k}^{l} \leq \mathit{Dep}_{max}^{l} \forall k \in S \left(l\right) , \forall l \in L \tag{eq\_0006} \\
  & t_{k l l^{'}}^{R} = \mathit{Dep}_{k}^{l^{'}} - \mathit{Arr}_{k}^{l} - \mathit{Tra}_{k}^{l l^{'}} \forall k \in S \left(l\right) \cap S \left(l^{'}\right) , \forall l , l^{'} \in L \tag{eq\_0007} \\
  & M \cdot \left(x_{l l^{'}}^{k} - 1\right) \leq t_{k l l^{'}}^{R} \leq M \cdot x_{l l^{'}}^{k} \forall k \in S \left(l\right) \cap S \left(l^{'}\right) , \forall l , l^{'} \in L \tag{eq\_0009} \\
  & \sum_{\forall l \in L} \sum_{\forall l^{'} \in L} \sum_{\forall i \in S \left(l\right)} \sum_{\forall j \in S \left(l^{'}\right)} \mathit{alpha}_{l l^{'}}^{i j} \leq N \forall l \neq l^{'} , \forall i \neq j \tag{eq\_0010} \\
  & x_{l l^{'}}^{k} + \mathit{alpha}_{l l^{'}}^{i j} \leq 1 \forall l , l^{'} \in L , \forall k \in S \left(l\right) \cap S \left(l^{'}\right) , \forall i \in S \left(l\right) , \forall j \in S \left(l^{'}\right) \tag{eq\_0011} \\
  & C \left[k\right]_{l} - C \left[i\right]_{l} > 0 \forall i , k \in S \left(l\right) , \forall l \in L \tag{eq\_0012} \\

  & \sum_{\forall l \in L} \sum_{\forall l^{'} \in L} \sum_{\forall k \in S \left(l\right) \cap S \left(l^{'}\right)} \left(1 - x_{l l^{'}}^{k}\right) - \sum_{\forall l \in L} \sum_{\forall l^{'} \in L} \sum_{\forall i \in S \left(l\right)} \sum_{\forall j \in S \left(l^{'}\right)} \mathit{alpha}_{l l^{'}}^{i j} \geq 0 \tag{eq\_0014} \\

  & \sum_{i \in S \left(l\right)} \mathit{alpha}_{l l^{'}}^{i j} \leq 1 \forall j \in S \left(l^{'}\right) , \forall l , l^{'} \in L \tag{eq\_0016} \\

  & \sum_{i \in S \left(l\right)} \mathit{alpha}_{l l^{'}}^{i j} + \sum_{j \in S \left(l^{'}\right)} \mathit{alpha}_{l l^{'}}^{i j} \leq 1 \forall i \in S \left(l\right) , \forall j \in S \left(l^{'}\right) , \forall l , l^{'} \in L \tag{eq\_0040} \\

  & \mathit{alpha}_{l l^{'}}^{b j} - \mathit{alpha}_{l l^{'}}^{d j} \geq 0 \forall b , d \in S \left(l\right) , \forall j \in S \left(l^{'}\right) , \forall l , l^{'} \in L \tag{eq\_0043} \\
  & C \left[d\right]_{l} - C \left[b\right]_{l} > 0 \forall b , d \in S \left(l\right) , \forall l \in L \tag{eq\_0044} \\
