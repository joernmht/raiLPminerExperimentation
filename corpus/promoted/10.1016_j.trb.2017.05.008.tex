  \min\quad & \sum_{t \in T} y_{t} \tag{eq\_0007\_a} \\
  & y_{t} \geq \mathit{omega}_{t , e n d} - e_{t , s} - c_{t} - \mathit{dt}_{t} t \in T , s \in R_{t} \tag{eq\_0007\_b} \\
  & y_{t} \geq 0 t \in T \tag{eq\_0007\_c} \\
  & \mathit{omega}_{t,q} - \mathit{omega}_{t,p} \ge \mathit{delta}_{p} \qquad \forall t \in T, p \in S, q \in S, p \notin V, q \notin V, p \in R_{t}, q \in R_{t}, (p,q) \in F \tag{eq\_0008} \\
  & \mathit{omega}_{t,q} - \mathit{omega}_{t,p} \ge \mathit{delta}_{p} + \sum_{k \in B} deltat_{k} \cdot \mathit{beta}_{k,t,p} \qquad \forall t \in T, p \in V, q \in V, p \in R_{t}, q \in R_{t}, (p,q) \in F \tag{eq\_0009} \\
  & \mathit{omega}_{t,q} - \mathit{omega}_{t,p} < \mathit{delta}_{p} + \sum_{l \in B_2} dtprev_{l} \cdot \mathit{beta}_{l,t,p} + M \cdot \mathit{beta}_{k,t,p} \qquad \forall t \in T, p \in V, q \in V, p \in R_{t}, q \in R_{t}, (p,q) \in F, k \in B_1 \tag{eq\_0010} \\
  & \mathit{omega}_{t,q} - \mathit{omega}_{t,p} \le \mathit{delta}_{p} + deltat_{k} \qquad \forall t \in T, p \in S, q \in S, p \in R_{t}, q \in R_{t}, (p,q) \in sw, k \in B_1 \tag{eq\_0011} \\
  & \mathit{omega}_{t,s} \ge e_{t,s} + dt_{t} + omega0 \qquad \forall (t,s) \in E_T \tag{eq\_0012} \\
  & \mathit{omega}_{t,q} - \mathit{omega}_{t,p} \ge \mathit{delta}_{p} + d_{t,p} \qquad \forall t \in T, q \in R_{t}, p \in D_{t} \tag{eq\_0013} \\
  & \mathit{omega}_{t,r} \ge planned_{t,r} + omega0 \qquad \forall t \in T, r \in Dplus_{t} \tag{eq\_0014} \\
  & \sum_{k \in Bbar} \mathit{beta}_{k,t,s} = 1 \qquad \forall t \in T, s \in V, s \in R_{t} \tag{eq\_0015} \\
  & \mathit{beta}_{k,t,p} + \mathit{beta}_{l,t,q} \le 1 \qquad \forall t \in T, p \in V, q \in V, p \in R_{t}, q \in R_{t}, (p,q) \in F, (k,l) \in B_{pair} \tag{eq\_0016} \\
  & \mathit{beta}_{k,t,s} = 1 \qquad \forall t \in T, s \in V, s \in AD_{t}, k \in B_1 \tag{eq\_0017} \\
  & M \cdot x_{t,s} \ge \mathit{omega}_{t,s} - W_{1} \qquad \forall t \in T, s \in Sbar, s \in R_{t} \tag{eq\_0018} \\
  & \left(\bar{x}\right)_{t , s} \geq \frac{W_{2} - \mathit{omega}_{t , s}}{M} \forall t \in T , s \in \bar{S} \cap R_{t} \tag{eq\_0019} \\
  & \sum_{\mathit{xi} = 1}^{B_{m}} \mathit{beta}_{\mathit{xi} , t , s} \geq x_{t , s} + \left(\bar{x}\right)_{t , s} - 1 \forall t \in T , s \in \bar{S} \cap R_{t} , \mathit{xi} \in B \tag{eq\_0020} \\
  & \mathit{omega}_{j , q} - \mathit{omega}_{i , p} \geq \mathit{delta}_{p , q} - \left(1 - \mathit{lambda}_{i , j , p , q}\right) M \forall i , j \in T , \left(\left(p , q\right) , \left(h , k\right)\right) \in A , p , k \in R_{i} , q , h \in R_{j} , i \neq j \tag{eq\_0021} \\
  & \mathit{omega}_{i , k} - \mathit{omega}_{j , h} \geq \mathit{delta}_{h , k} - \mathit{lambda}_{i , j , p , q} M \forall i , j \in T , \left(\left(p , q\right) , \left(h , k\right)\right) \in A , p , k \in R_{i} , q , h \in R_{j} , i \neq j \tag{eq\_0022} \\
  & \begin{matrix}\mathit{omega}_{j , q} - \mathit{omega}_{i , p} \geq - \left(2 - \mathit{mu}_{i , j , \mathit{vartheta}_{\mathit{eta}}} - \mathit{beta}_{\mathit{xi} , j , q}\right) M \\ \forall i , j \in T , \mathit{xi} \in B , p , q \in \mathit{vartheta}_{\mathit{eta}} , q \in R_{j} , p \in R_{i} , i \neq j , p = q + \mathit{xi}\end{matrix} \tag{eq\_0023} \\
  & \begin{matrix}\mathit{omega}_{i , q} - \mathit{omega}_{j , p} \geq - \left(1 + \mathit{mu}_{i , j , \mathit{vartheta}_{\mathit{eta}}} - \mathit{beta}_{\mathit{xi} , i , q}\right) M \\ \forall i , j \in T , \mathit{xi} \in B , p , q \in \mathit{vartheta}_{\mathit{eta}} , q \in R_{j} , p \in R_{i} , i \neq j , p = q + \mathit{xi}\end{matrix} \tag{eq\_0024} \\
  & M \mathit{mu}_{i , j , \mathit{vartheta}_{\mathit{eta}}} \geq \mathit{omega}_{j , f_{\mathit{eta}}} - \mathit{omega}_{i , f_{\mathit{eta}}} \forall i , j \in T , f_{\mathit{eta}} \in R_{i} , f_{\mathit{eta}} \in R_{j} \tag{eq\_0025} \\
  & \mathit{beta}_{\mathit{xi} , t , s} = 1 \forall t \in T - T_{0} , s \in \cup G_{\mathit{vartheta}} , \mathit{xi} = \bar{B_{m}} , s \in R_{t} \tag{eq\_0026} \\

  & \mathit{beta}_{\mathit{xi} , t , s} = 1 \forall t \in T_{0} , s \in \bar{S} , \mathit{xi} = B_{m} , s \in R_{t} \tag{eq\_0028} \\
