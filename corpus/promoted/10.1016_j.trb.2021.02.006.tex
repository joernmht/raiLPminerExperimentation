  & y_{i j} = \mathit{pi}_{j} - \mathit{pi}_{i} + T \cdot p_{i j} \forall \left(i , j\right) \in A \tag{eq\_0012} \\
  & \mathit{alpha}_{v , vp} + \mathit{alpha}_{vp , v} = 1 \forall k \in \mathcal{OD} , v \in V^{k} , vp \in V^{k} \backslash \left\{v\right\} \tag{eq\_0016} \\
  & W_{v}^{k} = \frac{1}{2} L_{v}^{k} \forall k \in \mathcal{OD} , \forall v \in V^{k} \tag{eq\_0017} \\
  & L_{v}^{k} \in \left[0 , T\right] \forall k \in \mathcal{OD} , v \in V^{k} \tag{eq\_0019} \\
  & Q_{v , vp} = \mathit{pi}_{vp} - \mathit{pi}_{v} + T \cdot \mathit{alpha}_{v , vp} . \tag{eq\_0025} \\
  & L_{v}^{k} \leq Q_{vp , v} \forall k \in \mathcal{OD} , vp \in V^{k} \backslash \left\{v\right\} \tag{eq\_0026} \\
  & \sum_{v \in V^{k}} L_{v}^{k} = T . \forall k \in \mathcal{OD} \tag{eq\_0027} \\
  & x_{v , d}^{k} \leq x_{v , d - 1}^{k} \forall k \in \mathcal{OD} , v \in V^{k} , d \in \left\{2 , \ldots , T\right\} . \tag{eq\_0036} \\
  & L_{v}^{k} = \sum_{d = 1}^{T} x_{v , d}^{k} \tag{eq\_0037\_a} \\
  & Y_{v}^{k} \leq Y_{r} + \mathit{gamma}_{w} \cdot Q_{v , vp} \forall vp \in V^{k} , r \in \mathcal{R}_{vp}^{k} \tag{eq\_0043} \\
  & Y_{v}^{k} \geq Y_{r} + \mathit{gamma}_{w} \cdot Q_{v , vp} - M_{v}^{k} \cdot \left(1 - z_{v , vp , r}^{k}\right) \forall vp \in V^{k} , r \in \mathcal{R}_{vp}^{k} \tag{eq\_0044} \\
  & \sum_{vp \in V^{k}} \underset{r \in \mathcal{R}_{vp}^{k}}{\sum} z_{v , vp , r}^{k} = 1 . \tag{eq\_0045} \\
  & \sum_{v \in V^{k}} L_{v}^{k} = T \tag{eq\_0049} \\
\end{align}
