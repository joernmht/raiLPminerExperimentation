  \min\quad & \sum_{i \in T} z_{n_{i}} \tag{eq\_0002} \\
  & x_{k}^{\text{end}} = x_{k + 1}^{\text{begin}} , k \in K_{i} , i \in T : k \neq n_{i} , \tag{eq\_0004} \\
  & xend_{k} \geq xbegin_{k} + d_{k} \qquad \forall k \in E \tag{eq\_0005} \\

  & x_{k}^{\text{begin}} = b_{k}^{\text{static}} , k \in E : b_{k}^{\text{static}} > 0 , \tag{eq\_0007} \\

  & xend_{k} - einitial_{k} \leq z_{k} \qquad \forall k \in E \tag{eq\_0009} \\
  & \sum_{t \in P_{j}} q_{k,t} = 1 \qquad \forall k \in L_{j}, j \in B \tag{eq\_0010} \\
  & q_{k,t} + q_{khat,t} - 1 \leq \mathit{lambda}_{k,khat} + \mathit{gamma}_{k,khat} \qquad \forall k, khat \in L_{j}, t \in P_{j}, j \in B : k < khat \tag{eq\_0011} \\
  & xbegin_{khat} - xend_{k} \geq \mathit{Delta}_{j}^{M} \cdot \mathit{gamma}_{k,khat} - M \cdot (1 - \mathit{gamma}_{k,khat}) \qquad \forall k, khat \in L_{j}, j \in B : k < khat, o_{khat} \neq o_{k} \tag{eq\_0012} \\
  & xbegin_{khat} - xend_{k} \geq \mathit{Delta}_{j}^{F} \cdot \mathit{gamma}_{k,khat} - M \cdot (1 - \mathit{gamma}_{k,khat}) \qquad \forall k, khat \in L_{j}, j \in B : k < khat, o_{khat} = o_{k} \tag{eq\_0013} \\
  & xbegin_{k} - xend_{khat} \geq \mathit{Delta}_{j}^{M} \cdot \mathit{lambda}_{k,khat} - M \cdot (1 - \mathit{lambda}_{k,khat}) \qquad \forall k, khat \in L_{j}, j \in B : k < khat, o_{khat} \neq o_{k} \tag{eq\_0014} \\
  & xbegin_{k} - xend_{khat} \geq \mathit{Delta}_{j}^{F} \cdot \mathit{lambda}_{k,khat} - M \cdot (1 - \mathit{lambda}_{k,khat}) \qquad \forall k, khat \in L_{j}, j \in B : k < khat, o_{khat} = o_{k} \tag{eq\_0015} \\
  & \mathit{lambda}_{k,khat} + \mathit{gamma}_{k,khat} \leq 1 \qquad \forall k, khat \in L_{j}, j \in B : k < khat \tag{eq\_0016} \\
  & gtrain_{i} \cdot q_{k,t} \leq gtrack_{j,t} \qquad \forall k \in (K_{i} \cap L_{j}), t \in P_{j}, j \in B, i \in T \tag{eq\_0017} \\
