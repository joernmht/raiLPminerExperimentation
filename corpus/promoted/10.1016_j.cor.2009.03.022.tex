  & \mathit{ell}_{a} \leq \mathit{pi}_{j} - \mathit{pi}_{i} \leq u_{a} \forall a = \left(i , j\right) \in A . \tag{eq\_0001} \\
  & \mathit{pi}_{j} \in \left[\mathit{pi}_{i} + \mathit{ell}_{a} , \mathit{pi}_{i} + u_{a}\right] . \tag{eq\_0002} \\
  & A = A_{drive} \cup A_{stop} \cup A_{transfer} \cup A_{turn} \cup A_{head} . \tag{eq\_0003} \\
  & \mathit{ell}_{a} \leq \mathit{pi}_{j} - \mathit{pi}_{i} \leq - \mathit{ell}_{a^{- 1}} . \tag{eq\_0004} \\
  & \underset{̲}{D} = \left(\underset{̲}{V} , \underset{̲}{A}\right) \tag{eq\_0005} \\
  & \forall a = \left(i , j\right) \in \underset{̲}{A} \exists k_{a} \in \mathcal{Z} : \mathit{ell}_{a} \leq \mathit{pi}_{j} - \mathit{pi}_{i} + \mathit{Tk}_{a} \leq u_{a} . \tag{eq\_0006} \\
  & \underset{̲}{A} = \underset{̲}{A}_{drive} \cup \underset{̲}{A}_{stop} \cup \underset{̲}{A}_{transfer} \cup \underset{̲}{A}_{turn} \cup \underset{̲}{A}_{head} . \tag{eq\_0007} \\
  & \left(PESP - IP - \mathit{pi}\right) min f \left(\mathit{pi} , p\right) = \sum_{a = \left(i , j\right) \in \underset{̲}{A}} {w}_{a} \left(\mathit{pi}_{j} - \mathit{pi}_{i} + \mathit{Tk}_{a} - \mathit{ell}_{a}\right) \tag{eq\_0009} \\
  & \mathit{pi}_{j} - \mathit{pi}_{i} + \mathit{Tk}_{a} \leq u_{a} for all a = \left(i , j\right) \in \underset{̲}{A} , \tag{eq\_0010} \\
  & \mathit{pi}_{j} - \mathit{pi}_{i} + \mathit{Tk}_{a} \geq \mathit{ell}_{a} for all a = \left(i , j\right) \in \underset{̲}{A} , \tag{eq\_0011} \\
  & \mathit{pi}_{i} \geq 0 for all i \in \underset{̲}{V} , \tag{eq\_0012} \\
  & \mathit{pi}_{i} < T for all i \in \underset{̲}{V} , \tag{eq\_0013} \\
  & k_{a} \in \mathcal{Z} for all a \in \underset{̲}{A} . \tag{eq\_0014} \\
  & u_{a} - \mathit{ell}_{a} < T . \tag{eq\_0016} \\
  & \mathit{pi}_{first} \left(i\right) = min \left\{\mathit{pi}_{i} + \mathit{kT} : \mathit{pi}_{i} + \mathit{kT} \geq t_{min} , k \in \mathcal{Z}\right\} \tag{eq\_0017\_a} \\
  & \mathit{pi}_{last} \left(i\right) = max \left\{\mathit{pi}_{i} + \mathit{kT} : \mathit{pi}_{i} + \mathit{kT} \leq t_{max} , k \in \mathcal{Z}\right\} \tag{eq\_0017\_b} \\
  & z_{a} = \left\{\begin{matrix} 0 & if transfer activity a is maintained , \\ 1 & otherwise \end{matrix}\right \tag{eq\_0020} \\
  & g_{a} = \left\{\begin{matrix} 0 & if i starts \cdot before j , \\ 1 & otherwise . \end{matrix}\right \tag{eq\_0021} \\
  \min\quad & f \left({\mathit{pi}} , z\right) = \sum_{i \in V} w_{i} \left({\mathit{pi}}_{i} - \mathit{pi}_{i}\right) + \sum_{a \in A_{transfer}} w_{a} \cdot \mathit{Tz}_{a} \tag{eq\_0022} \\
  & {\mathit{pi}}_{i} \geq \mathit{pi}_{i} for all i \in V , \tag{eq\_0023} \\
  & {\mathit{pi}}_{j} - {\mathit{pi}}_{i} \geq \mathit{ell}_{a} + d_{a} for all a = \left(i , j\right) \in A_{stop} \cup A_{drive} \cup A_{turn} , \tag{eq\_0024} \\
  & \mathit{Mz}_{a} + {\mathit{pi}}_{j} - {\mathit{pi}}_{i} \geq \mathit{ell}_{a} for all a = \left(i , j\right) \in A_{transfer} , \tag{eq\_0025} \\
  & \mathit{Mg}_{a} + {\mathit{pi}}_{j} - {\mathit{pi}}_{i} \geq \mathit{ell}_{a} for all a = \left(i , j\right) \in A_{head} , \tag{eq\_0026} \\
  & g_{a} + g_{a^{- 1}} = 1 for all a , a^{- 1} \in A_{head} , \tag{eq\_0027} \\
  & {\mathit{pi}}_{i} \in \mathcal{N} for all i \in V , z_{a} \in \left\{0 , 1\right\} for all a \in A_{transfer} , g_{a} \in \left\{0 , 1\right\} for all a \in A_{head} . \tag{eq\_0028} \\
  & PoR \left(ID\right) := C_{DEF} \left(ID\right) / C_{DEF} \left(DEF\right) , \tag{eq\_0030} \\
  & RoD \left(ID\right) = C_{ID}^{'} \left(DEF\right) / C_{ID}^{'} \left(ID\right) . \tag{eq\_0031} \\
