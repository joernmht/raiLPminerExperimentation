\begin{align}
  & D_{l , i , j} = A_{l , i , j} + S_{l , i , j} , \forall l \in L , i \in I_{l} , j \in J_{l} \tag{eq\_0001} \\
  & A_{l,i,j} = D_{l,i-1,j} + R_{l,i-1,j} \qquad \forall l \in L, i \in I_{l} \setminus \{1\}, j \in J_{l} \tag{eq\_0003} \\
  & H_{l,i,j} = D_{l,i,j} - D_{l,i,j-1} \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \setminus \{1\} \tag{eq\_0004} \\
  & d_{l , i , j} = a_{l , i , j} + s_{l , i , j} , \forall l \in L , i \in I_{l} , j \in J_{l} \tag{eq\_0005} \\
  & a_{l,i,j} = A_{l,i,j} \qquad \forall l \in L, i \in \{1\}, j \in J_{l} \tag{eq\_0007\_a} \\
  & a_{l,i,j} = d_{l,i-1,j} + r_{l,i-1,j} \qquad \forall l \in L, i \in I_{l} \setminus \{1\}, j \in J_{l} \tag{eq\_0007\_b} \\
  & s_{l , i , j} = S_{l , i , j} + u_{l , i , j}^{s} + w_{l , i , j}^{s} , \forall l \in L , i \in I_{l} , j \in J_{l} \tag{eq\_0008} \\
  & r_{l,i,j} = R_{l,i,j} + u_{l,i,j}^{r} + w_{l,i,j}^{r} \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \setminus \{N_{l}\} \tag{eq\_0009} \\
  & us_{l}^{min} \le u_{l,i,j}^{s} \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \tag{eq\_0010\_a} \\
  & u_{l,i,j}^{s} \le us_{l}^{max} \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \tag{eq\_0010\_b} \\
  & ur_{l}^{min} \le u_{l,i,j}^{r} \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \setminus \{N_{l}\} \tag{eq\_0012\_a} \\
  & u_{l,i,j}^{r} \le ur_{l}^{max} \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \setminus \{N_{l}\} \tag{eq\_0012\_b} \\
  & h_{l,i,j} = d_{l,i,j} - d_{l,i,j-1} \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \setminus \{1\} \tag{eq\_0013} \\
  & h_{l,i,j} \ge hmin_{l} \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \setminus \{1\} \tag{eq\_0014} \\
  & p_{l,i,j}^{wp} = \mathit{lambda}_{l,i,j} \cdot (d_{l,i,j} - T_{0}) \qquad \forall l \in L, i \in I_{l}, j \in \{1\} \tag{eq\_0015\_a} \\
  & p_{l,i,j}^{wp} = \mathit{lambda}_{l,i,j} \cdot h_{l,i,j} + p_{l,i,j-1}^{sp} \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \setminus \{1\} \tag{eq\_0015\_b} \\
  & p_{l , i , j}^{s p} = p_{l , i , j}^{w p} - p_{l , i , j}^{e p} , \forall l \in L , i \in I_{l} , j \in J_{l} \tag{eq\_0016} \\
  & p_{l,i,j}^{ep} \ge 0 \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \tag{eq\_0017} \\
  & p_{l,i,j}^{ep} \le p_{l,i,j}^{wp} \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \tag{eq\_0019} \\
  & p_{l,i,j}^{wt} \le cp_{l,i} \qquad \forall l \in L, i \in I_{l}, j \in J_{l} \tag{eq\_0020} \\
  & p_{l , i , j}^{w t} = \left\{\begin{matrix} p_{l , i , j}^{e p} + \mathit{delta}_{l , i} \sum_{l^{'} \in X_{l , i}} \underset{j^{'} \in J_{l^{'}}}{\sum} g_{l , i , j}^{l^{'} , j^{'}} , \forall l \in L , i \in I_{l} , j = 1 \\ p_{l , i , j}^{e p} + \mathit{delta}_{l , i} \sum_{l^{'} \in X_{l , i}} \underset{j^{'} \in J_{l^{'}}}{\sum} g_{l , i , j}^{l^{'} , j^{'}} + p_{l , i , j - 1}^{s t} , \forall l \in L , i \in I_{l} , j \in J_{l} \backslash \left\{1\right\} \end{matrix}\right \tag{eq\_0021} \\
  & d_{l , i , j} - \left(a_{l^{'} , i , j^{'}} + o_{l , i}^{l^{'}}\right) \geq M \left(y_{l , i , j}^{l^{'} , j^{'}} - 1\right) , \forall l \in L , i \in U_{l} , l^{'} \in X_{l , i} , j \in J_{l} , j^{'} \in J_{l^{'}} \tag{eq\_0022} \\
  & d_{l , i , j} - \left(a_{l^{'} , i , j^{'}} + o_{l , i}^{l^{'}}\right) \leq M \cdot y_{l , i , j}^{l^{'} , j^{'}} , \forall l \in L , i \in U_{l} , l^{'} \in X_{l , i} , j \in J_{l} , j^{'} \in J_{l^{'}} \tag{eq\_0024} \\
  & z_{l , i , j}^{l^{'} , j^{'}} = \left\{\begin{matrix} y_{l , i , j}^{l^{'} , j^{'}} , \forall l \in L , i \in U_{l} , j = 1 , l^{'} \in X_{l , i} , j^{'} \in J_{l^{'}} \\ y_{l , i , j}^{l^{'} , j^{'}} - y_{l , i , j - 1}^{l^{'} , j^{'}} , \forall l \in L , i \in U_{l} , j \in J_{l} \backslash \left\{1\right\} , l^{'} \in X_{l , i} , j^{'} \in J_{l^{'}} \end{matrix}\right \tag{eq\_0025} \\
  & g_{l , i , j}^{l^{'} , j^{'}} = \mathit{alpha}_{l , i}^{l^{'} , j^{'}} z_{l , i , j}^{l^{'} , j^{'}} p_{l^{'} , i , j^{'}}^{a l} , \forall l \in L , i \in U_{l} , l^{'} \in X_{l , i} , j \in J_{l} , j^{'} \in J_{l^{'}} \tag{eq\_0026} \\
  & p_{l , i , j}^{s t} = p_{l , i , j}^{w t} - p_{l , i , j}^{b t} , \forall l \in L , i \in I_{l} , j \in J_{l} \tag{eq\_0027} \\
  & p_{l , i , j}^{b t} = \left\{\begin{matrix} min \left\{p_{l , i , j}^{w t} , c \cdot t_{l , j}\right\} , \forall l \in L , i = 1 , j \in J_{l} \\ min \left\{p_{l , i , j}^{w t} , c \cdot t_{l , j} - p_{l , i - 1 , j}^{o n} + p_{l , i , j}^{a l}\right\} , \forall l \in L , i \in I_{l} \backslash \left\{1\right\} , j \in J_{l} \end{matrix}\right \tag{eq\_0029} \\
  & p_{l , i , j}^{o n} = \left\{\begin{matrix} p_{l , i , j}^{b t} , \forall l \in L , i = 1 , j \in J_{l} \\ p_{l , i - 1 , j}^{o n} + p_{l , i , j}^{b t} - p_{l , i , j}^{a l} , \forall l \in L , i \in I_{l} \backslash \left\{1\right\} , j \in J_{l} \end{matrix}\right \tag{eq\_0030} \\
  & p_{l , i , j}^{a l} = \left\{\begin{matrix} 0 , \forall l \in L , i = 1 , j \in J_{l} \\ \mathit{beta}_{l , i , j} p_{l , i - 1 , j}^{o n} , \forall l \in L , i \in I_{l} \backslash \left\{1\right\} , j \in J_{l} \end{matrix}\right \tag{eq\_0031} \\
  \min\quad & F \tag{eq\_0032\_a} \\
  & F = F1 + F2 + F3 + F4 \tag{eq\_0032\_b} \\
  & F_{1} = \sum_{l \in L} \sum_{i \in I_{l}} \sum_{j \in J_{l}} \left[\mathit{xi}_{1} \left(d_{l , i , j} - D_{l , i , j}\right)^{2} + \mathit{xi}_{2} \left(h_{l , i , j} - H_{l , i , j}\right)^{2}\right] \tag{eq\_0033} \\
  & F_{2} = \sum_{l \in L} \sum_{i \in I_{l}} \sum_{j \in J_{l}} \begin{bmatrix} \mathit{xi}_{3} \frac{1}{2} \mathit{lambda}_{l , i , j} h_{l , i , j}^{2} + \mathit{xi}_{4} p_{l , i , j - 1}^{s t} h_{l , i , j} + \mathit{xi}_{5} p_{l , i , j - 1}^{s p} h_{l , i , j} \\ + \mathit{xi}_{6} \mathit{delta}_{l , i} \sum_{l^{'} \in X_{l , i}} \underset{j^{'} \in J_{l^{'}}}{\sum} g_{l , i , j}^{l^{'} , j^{'}} \left(d_{l , i , j} - a_{l^{'} , i , j^{'}} - o_{l , i}^{l^{'}}\right) \end{bmatrix} \tag{eq\_0034} \\
  & F_{3} = \sum_{l \in L} \sum_{i \in I_{l}} \sum_{j \in J_{l}} \mathit{xi}_{7} p_{l , i , j}^{a r} \tag{eq\_0035} \\
  & F_{4} = \sum_{l \in L} \sum_{i \in I_{l}} \sum_{j \in J_{l}} \left[\mathit{xi}_{8} \left(u_{l , i , j}^{r}\right)^{2} + \mathit{xi}_{9} \left(u_{l , i , j}^{s}\right)^{2}\right] \tag{eq\_0037} \\
  & y_{l , i , j}^{l^{'} , j^{'}} = \left\{0 , 1\right\} , \forall l \in L , i \in U_{l} , l^{'} \in X_{l , i} , j \in J_{l} , j^{'} \in J_{l^{'}} \tag{eq\_0041} \\
  & \mathit{ur}_{l}^{min} \leq u_{l , i , j}^{r} \leq \mathit{ur}_{l}^{max} , \forall l \in L , i \in I_{l} , j \in J_{l} \tag{eq\_0042} \\
  & \mathit{us}_{l}^{min} \leq u_{l , i , j}^{s} \leq \mathit{us}_{l}^{max} , \forall l \in L , i \in I_{l} , j \in J_{l} \tag{eq\_0043} \\
  & p_{l , i , j}^{e p} \geq 0 , \forall l \in L , i \in I_{l} , j \in J_{l} \tag{eq\_0044} \\
\end{align}
