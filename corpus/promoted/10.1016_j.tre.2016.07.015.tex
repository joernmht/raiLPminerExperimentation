  \begin{aligned}\quad & & min \\ & \sum_{t \in T} \mathit{gamma}_{t} y_{t} + \sum_{e \in E} \mathit{mu}_{e}^{+} d_{e}^{+} + \underset{e \in E^{\mathit{arr}}}{\sum} \mathit{mu}_{e}^{-} d_{e}^{-}\end{aligned} \tag{eq\_0001} \\

  & d_{e}^{+} \geq x_{e} - q_{e} - M_{1} \cdot y_{t_{e}} \forall e \in E \tag{eq\_0004} \\
  & d_{e}^{-} \geq q_{e} - x_{e} \forall e \in E^{\mathit{arr}} \tag{eq\_0005} \\
  & d_{e}^{+} \leq D_{e} \forall e \in E \tag{eq\_0006} \\
  & d_{e}^{-} \leq D_{e} \forall e \in E^{\mathit{arr}} \tag{eq\_0007} \\
  & x_{e} = q_{e} \forall e \in E : q_{e} < H_{\mathit{dis}}^{\mathit{start}} \tag{eq\_0008} \\
  & y_{t_{e}} = 0 \forall t_{e} \in T , e \in E_{s_{t_{e}}^{o}}^{\mathit{dep}} : q_{e} < H_{\mathit{dis}}^{\mathit{start}} \tag{eq\_0009} \\
  & y_{t} \in \left\{0 , 1\right\} \forall t \in T \tag{eq\_0010} \\
  & x_{e} , d_{e}^{+} , d_{e}^{-} \geq 0 \forall e \in E \tag{eq\_0011} \\
  & x_{f} - x_{e} \geq L_{a} \forall a = \left(e , f\right) \in A_{\mathit{train}} \tag{eq\_0012} \\
  & \mathit{lambda}_{a} + \mathit{lambda}_{a^{'}} = 1 \forall a = \left(e , f\right) \in A_{\mathit{head}}^{\mathit{track} , 1} \land a^{'} = \left(f , e\right) \in A_{\mathit{head}}^{\mathit{track} , 1} \tag{eq\_0014} \\
  & x_{f} - x_{e} + M_{2} \cdot \left(1 - \mathit{lambda}_{a}\right) \geq L_{a} \forall a = \left(e , f\right) \in A_{\mathit{head}}^{\mathit{track} , 1} \tag{eq\_0015} \\
  & H_{\mathit{dis}}^{\mathit{end}} \cdot \left(1 - \mathit{chi}_{e}^{d}\right) < x_{e} \leq H_{\mathit{dis}}^{\mathit{end}} + \left(1 - \mathit{chi}_{e}^{d}\right) \cdot M_{2} \tag{eq\_0017} \\
  & \mathit{lambda}_{a} + \mathit{lambda}_{a^{'}} = 1 \forall a = \left(e , f\right) \in A_{\mathit{head}}^{\mathit{track} , 2} \land a^{'} = \left(f , e\right) \in A_{\mathit{head}}^{\mathit{track} , 2} \tag{eq\_0018} \\
  & x_{f} - x_{e} + M_{2} \cdot \left(1 - \mathit{lambda}_{a}\right) + M_{2} \cdot \left(1 - \mathit{chi}_{e_{1}}^{d}\right) \geq L_{a} \forall a = \left(e , f\right) \in A_{\mathit{head}}^{\mathit{track} , 2} , e_{1} \in E^{\mathit{dep}} \tag{eq\_0019} \\
  & \mathit{lambda}_{a} - \mathit{lambda}_{a^{'}} = 0 \forall \left(a , a^{'}\right) \in B \tag{eq\_0020} \\
  & \sum_{a = \left(e , f\right) \in A_{s , f}^{1}} \mathit{lambda}_{a} - \sum_{a = \left(e , f\right) \in A_{s , f}^{2}} \mathit{varphi}_{a} \leq C_{s}^{\mathit{down}} - 1 \forall s \in {S} , f \in E_{s}^{\mathit{arr}} : q_{f} \geq H_{\mathit{dis}}^{\mathit{start}} \land t_{f} \in T^{\mathit{down}} \tag{eq\_0022} \\

  & \sum_{t \in p} y_{t} - \sum_{t \in p^{'}} y_{t} \leq \mathit{delta} \forall \left(p , p^{'}\right) \in \mathit{PP} \tag{eq\_0025} \\

  & x_{f} - x_{e} \leq L_{a}^{\mathit{max}} \forall a = \left(e , f\right) \in A_{\mathit{train}}^{\mathit{run}} , s_{f} \geq s_{l} , t_{e} \in T_{\mathit{dis}}^{\mathit{up}} \tag{eq\_0031} \\

  & \sum_{a = \left(e , f\right) \in A_{s , f}^{1}} \mathit{lambda}_{a} - \sum_{a = \left(e , f\right) \in A_{s , f}^{2}} \mathit{varphi}_{a} + \left|\mathit{omega}_{s , h_{0}}^{\mathit{down}}\right| \leq C_{s}^{\mathit{down}} - 1 \forall s \in {S} , f \in E_{s}^{\mathit{arr}} : q_{f} \geq H_{\mathit{dis}}^{\mathit{start}} \land t_{f} \in T^{\mathit{down}} \tag{eq\_0039} \\
  & \sum_{a = \left(e , f\right) \in A_{s , f}^{1}} \mathit{lambda}_{a} - \sum_{a = \left(e , f\right) \in A_{s , f}^{2}} \mathit{varphi}_{a} + \left|\mathit{omega}_{s , h_{0}}^{\mathit{up}}\right| \leq C_{s}^{\mathit{up}} - 1 \forall s \in {S} , f \in E_{s}^{\mathit{arr}} : q_{f} \geq H_{\mathit{dis}}^{\mathit{start}} \land t_{f} \in T^{\mathit{up}} \tag{eq\_0040} \\
