  & \frac{\mathit{gamma}_{e_{a}}^{1} + \mathit{gamma}_{e_{d}}^{2}}{2} - 0 . 5 \leq \mathit{gamma}_{q} \forall s_{v} \in \mathit{Gamma} , \forall e_{a} \in E_{a r r}^{s_{v}} , \forall e_{d} \in E_{d e p}^{s_{v}} , q = \left(e_{a} , e_{d}\right) \in A_{s_{v}} \tag{eq\_0011\_c} \\
