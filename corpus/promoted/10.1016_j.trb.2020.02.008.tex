  & x_{v}^{l} = \sum_{a \in \mathit{eta}_{l,v}^{-}} y_{a}^{l} \qquad \forall l \in T \cup M, v \in \mathcal{V} \tag{eq\_0001} \\
  & \sum_{a \in \mathit{eta}_{l,v}^{+}} y_{a}^{l} - \sum_{a \in \mathit{eta}_{l,v}^{-}} y_{a}^{l} = 1 \qquad \forall l \in T \cup M, v \in \{\mathit{sigma}^{l}\} \tag{eq\_0002\_a} \\
  & \sum_{a \in \mathit{eta}_{l,v}^{+}} y_{a}^{l} - \sum_{a \in \mathit{eta}_{l,v}^{-}} y_{a}^{l} = -1 \qquad \forall l \in T \cup M, v \in \{\mathit{tau}^{l}\} \tag{eq\_0002\_b} \\
  & \sum_{a \in \mathit{eta}_{l,v}^{+}} y_{a}^{l} - \sum_{a \in \mathit{eta}_{l,v}^{-}} y_{a}^{l} = 0 \qquad \forall l \in T \cup M, v \in \mathcal{V} \setminus \{\mathit{sigma}^{l}, \mathit{tau}^{l}\} \tag{eq\_0002\_c} \\
  & y_{a}^{r} - y_{b}^{r} = 0 \qquad \forall r \in M, (a, b) \in R_{m}^{r} \tag{eq\_0003} \\
  & \sum_{v \in R_{1,w}} (x_{v}^{k} + x_{v}^{kp}) \leq 1 \qquad \forall k \in T, kp \in T \setminus \{k\}, i \in N, w \in W^{i} \tag{eq\_0004} \\
  & \sum_{v \in R_{2,u}} (x_{v}^{k} + x_{v}^{kp}) \leq 1 \qquad \forall k \in T, kp \in T \setminus \{k\}, i \in N, u \in U^{i} \tag{eq\_0005} \\
  & x_{w1}^{k} + x_{u1}^{k} + x_{w2}^{kp} + x_{u2}^{kp} \leq 3 \tag{eq\_0006} \\
  & \sum_{v \in W^{i} \cap \mathcal{V}^{k} : v \preceq w} x_{v}^{k} + \sum_{v \in U^{i + 1} \cap \mathcal{V}^{k} : v \succeq u} x_{v}^{k} + \underset{v \in W^{i} \cap \mathcal{V}^{k^{^{'}}} : v \succeq w}{\sum} x_{v}^{k^{^{'}}} + \underset{v \in U^{i + 1} \cap \mathcal{V}^{k^{^{'}}} : v \preceq u}{\sum} x_{v}^{k^{^{'}}} \leq 3 \tag{eq\_0007} \\
  & \sum_{k \in T, a \in \mathit{varphi}_{i,t}} y_{a}^{k} \leq B_{i}^{u} \qquad \forall i \in N, t \in H \tag{eq\_0008} \\
  & y_{a}^{r} + \sum_{k \in T, v \in R_{3,a}} x_{v}^{k} \leq 1 \qquad \forall r \in M, a \in \mathcal{A}^{r} \tag{eq\_0009} \\
  & y_{a}^{r} + \sum_{k \in T, v \in R_{4,a}} x_{v}^{k} \leq 1 \qquad \forall r \in M, a \in \mathcal{A}^{r} \tag{eq\_0010} \\
  & \sum_{k \in T} \underset{a \in \mathbf{\mathcal{A}}^{k}}{\sum} C_{a}^{k} \cdot Q_{a}^{k} \cdot y_{a}^{k} \tag{eq\_0011\_a} \\
  & \sum_{r \in M} \underset{a \in \mathbf{\mathcal{A}}^{r}}{\sum} C_{a}^{r} \cdot y_{a}^{r} \tag{eq\_0011\_b} \\
  \min\quad & F \tag{eq\_0012} \\
  & \sum_{r \in M, a \in \mathcal{A}^{r}} C_{a}^{r} \cdot y_{a}^{r} \leq \mathit{omega} \tag{eq\_0013\_a} \\
  & \mathit{omega} = \mathit{rho} \cdot Fmin \tag{eq\_0013\_b} \\
\end{align}
