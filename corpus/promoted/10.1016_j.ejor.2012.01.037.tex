  \min\quad & F(X,Z) \tag{eq\_0002} \\
  & \text{subject to} \sum_{c \in C} X_{r , c} = 1 \forall r \in R \tag{eq\_0003} \\
  & X_{r , c} = \sum_{c^{'} \in C} Z_{r , r^{'} , c , c^{'}} \forall \left(r , r^{'}\right) \in T , c \in C \tag{eq\_0004} \\
  & X_{r^{'} , c^{'}} = \sum_{c \in C} Z_{r , r^{'} , c , c^{'}} \forall \left(r , r^{'}\right) \in T , c^{'} \in C \tag{eq\_0005} \\

  & D_{s , m} \geq i_{s , m}^{\infty} - I_{s , m}^{\infty} \forall s \in S , m \in M \tag{eq\_0008\_a} \\
  & D_{s , m} \geq I_{s , m}^{\infty} - i_{s , m}^{\infty} \forall s \in S , m \in M \tag{eq\_0008\_b} \\
  & \sum_{r \in R} \sum_{c \in C} w_{r , c} \cdot X_{r , c} + \sum_{r , r^{'} \in T} \sum_{c , c^{'} \in C \cdot C} \mathit{gamma}_{r , r^{'} , c , c^{'}} Z_{r , r^{'} , c , c^{'}} + \mathit{beta} \cdot \sum_{s \in S} \sum_{m \in M} D_{s , m} \tag{eq\_0009} \\
