\begin{align}
  \max\quad & \sum_{i \in I} \sum_{j \in I \left(i\right)} \sum_{b \in B^{i j}} N_{b}^{i j} \sum_{p = 1}^{f^{i}} \sum_{q = 1}^{f^{j}} Y_{p q b}^{i j} \tag{eq\_0001} \\
  & \begin{matrix}\text{s}.\text{t}. X_{1}^{i} \leq H^{i} \cdot i \in I\end{matrix} \tag{eq\_0002} \\
  & \begin{matrix}T - H^{i} \leq X_{f^{i}}^{i} \leq T \cdot i \in I\end{matrix} \tag{eq\_0003} \\
  & \begin{matrix}h^{i} \leq X_{p + 1}^{i} - X_{p}^{i} \leq H^{i} \cdot i \in I , p = 1 , \hdots , f^{i} - 1\end{matrix} \tag{eq\_0004} \\
  & \begin{matrix} & & \left(X_{q}^{j} + t_{b}^{j}\right) - \left(X_{p}^{i} + t_{b}^{i}\right) \geq w_{b} - M \left(1 - Y_{p q b}^{i j}\right) i \in I , j \in I \left(i\right) , b \in B^{i j} , \\ & & p = 1 , \hdots , f^{i} , q = 1 , \hdots , f^{j}\end{matrix} \tag{eq\_0005} \\
  & \begin{matrix} & & \left(X_{q}^{j} + t_{b}^{j}\right) - \left(X_{p}^{i} + t_{b}^{i}\right) \leq W_{b} + M \left(1 - Y_{p q b}^{i j}\right) i \in I , j \in I \left(i\right) , b \in B^{i j} , \\ & & p = 1 , \hdots , f^{i} , q = 1 , \hdots , f^{j}\end{matrix} \tag{eq\_0006} \\
  & X_{p}^{i} \in \mathcal{R} i \in I , p = 1 , \hdots , f^{i} \tag{eq\_0007\_a} \\
  & Y_{p q b}^{i j} \in \left\{0 , 1\right\} i \in I , j \in I \left(i\right) , b \in B^{i j} , p = 1 , \hdots , f^{i} , q = 1 , \hdots , f^{j} \tag{eq\_0007\_b} \\
  & \left(\left({X}\right)_{q b}^{j} + t_{b}^{j}\right) - \left(\left({X}\right)_{p b}^{i} + t_{b}^{i}\right) \geq \left(\bar{d}\right)_{p}^{i} + t_{b}^{i} - \left(\underset{̲}{d}\right)_{q}^{j} - t_{b}^{j} . \tag{eq\_0009} \\
  & \left(\left({X}\right)_{q b}^{j} + t_{b}^{j}\right) - \left(\left({X}\right)_{p b}^{i} + t_{b}^{i}\right) \leq \left(\bar{d}\right)_{q}^{j} + t_{b}^{j} - \left(\underset{̲}{d}\right)_{p}^{i} - t_{b}^{j} . \tag{eq\_0010} \\
  & \begin{matrix}\sum_{q = 1}^{f^{j}} Y_{p q b}^{i j} \leq 1 + \lfloor \frac{W_{b} - w_{b}}{h^{j}} \rfloor & i \in I , j \in I \left(i\right) , b \in B^{i j} , p \in \left\{1 , \hdots , f^{i}\right\}\end{matrix} \tag{eq\_0011} \\
  & \begin{matrix}\sum_{p = 1}^{f^{i}} Y_{p q b}^{i j} \leq 1 + \lfloor \frac{W_{b} - w_{b}}{h^{i}} \rfloor & i \in I , j \in I \left(i\right) , b \in B^{i j} , q \in \left\{1 , \hdots , f^{j}\right\} .\end{matrix} \tag{eq\_0012} \\
  & \begin{aligned}Y_{p q b}^{i j} & + & \sum_{q^{'} = q + 1}^{f^{j}} Y_{p q^{'} b}^{i j} + \sum_{p^{'} = p + 1}^{f^{i}} Y_{p^{'} q b}^{i j} \leq 1 \\ & + & \lfloor \frac{W_{b} - w_{b}}{min \left(h^{i} , h^{j}\right)} \rfloor \begin{matrix} i \in I , j \in I \left(i\right) , b \in B^{i j} , \\ p \in \left\{1 , \hdots , f^{i}\right\} , q \in \left\{1 , \hdots , f^{j}\right\}\end{matrix}\end{aligned} \tag{eq\_0014} \\
  & \begin{aligned}Y_{p q b}^{i j} & + & \sum_{q^{'} = 1}^{q - 1} Y_{p q^{'} b}^{i j} + \sum_{p^{'} = 1}^{p - 1} Y_{p^{'} q b}^{i j} \leq 1 \\ & + & \lfloor \frac{W_{b} - w_{b}}{min \left(h^{i} , h^{j}\right)} \rfloor \begin{matrix} i \in I , j \in I \left(i\right) , b \in B^{i j} , \\ p \in \left\{1 , \hdots , f^{i}\right\} , q \in \left\{1 , \hdots , f^{j}\right\} .\end{matrix}\end{aligned} \tag{eq\_0015} \\
  & \left({Y}\right)_{p q b}^{i j} + c \leq \sum_{p^{'} = 1}^{f^{i}} \left({Y}\right)_{p^{'} q b}^{i j} \leq 1 + \lfloor \frac{W_{b} - w_{b}}{h^{i}} \rfloor \leq 1 + \lfloor \frac{W_{b} - w_{b}}{min \left(h^{i} , h^{j}\right)} \rfloor . \tag{eq\_0017} \\
  & \left({Y}\right)_{p q b}^{i j} + r \leq \sum_{q^{'} = 1}^{f^{j}} \left({Y}\right)_{p q^{'} b}^{i j} \leq 1 + \lfloor \frac{W_{b} - w_{b}}{h^{j}} \rfloor \leq 1 + \lfloor \frac{W_{b} - w_{b}}{min \left(h^{i} , h^{j}\right)} \rfloor . \tag{eq\_0018} \\
  & \left({X}\right)_{q + 1}^{j} + t_{b}^{j} \leq \left({X}\right)_{p}^{i} + t_{b}^{i} + W_{b} - \left(r - 1\right) h^{j} . \tag{eq\_0019} \\
  & \left({X}\right)_{q}^{j} + t_{b}^{j} \geq \left({X}\right)_{p + 1}^{i} + t_{b}^{i} + w_{b} + \left(c - 1\right) h^{i} . \tag{eq\_0020} \\
  & \begin{aligned}h^{j} & \leq & \left(\left({X}\right)_{q + 1}^{j} + t_{b}^{j}\right) - \left(\left({X}\right)_{q}^{j} + t_{b}^{j}\right) \\ & \leq & \left({X}\right)_{p}^{i} - \left({X}\right)_{p + 1}^{i} + W_{b} - w_{b} - \left(r - 1\right) h^{j} - \left(c - 1\right) h^{i} .\end{aligned} \tag{eq\_0021} \\
  & h^{j} \leq - h^{i} + W_{b} - w_{b} - \left(r - 1\right) h^{j} - \left(c - 1\right) h^{i} , \tag{eq\_0022} \\
  & r + c \leq r + c \frac{h^{i}}{h^{j}} \leq \frac{W_{b} - w_{b}}{h^{j}} . \tag{eq\_0023} \\
  & \left({Y}\right)_{p q b}^{i j} + r + c \leq 1 + \frac{W_{b} - w_{b}}{h^{j}} , \tag{eq\_0024} \\
  & \sum_{\left(i , j , p , q , b\right) \in E} Y_{p q b}^{i j} \leq \mathit{gamma} \tag{eq\_0025} \\
  & \sum_{\left(i , j , p , q , b\right) \in E^{'}} Y_{p q b}^{i j} \leq \mathit{gamma} \left(1 - Y_{p_{0} q_{0} b_{0}}^{i_{0} j_{0}}\right) \tag{eq\_0026} \\
\end{align}
