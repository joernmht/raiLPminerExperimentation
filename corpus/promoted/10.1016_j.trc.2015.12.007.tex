  & \mathit{pi} \cdot k = \mathit{pi}^{max} \cdot k - \left[\mathit{alpha} \cdot k \cdot \sum_{i, s, t} x_{i , i}^{t , s} \cdot k\right] - \left[\mathit{beta} \cdot k \cdot \sum_{i} y_{o \left(k\right) , i} \cdot k\right] \forall k \tag{eq\_0001} \\
  \text{Z}_{1}\quad & = max \sum_{k} \mathit{pi} \cdot k \tag{eq\_0002} \\
  & \sum_{s, j} x_{i , j}^{t , s} \cdot k - \sum_{s, j} x_{j , i}^{s , t} \cdot k = \left\{ 1 & i = o \left(k\right) , t = t_{o} \left(k\right) \\ - 1 & i = q \left(k\right) , t = t_{q} \left(k\right) \\ 0 & \text{otherwise} \right \forall k , i , t \tag{eq\_0003} \\
  & y_{o \left(k\right) , i} \cdot k = 0 \cdot i \leq o \cdot k \text{or} i > q \cdot k \forall k \tag{eq\_0004} \\
  & \sum_{k} y_{i , j} \cdot k \geq v_{i , j} \forall i , j , i \neq j \tag{eq\_0005} \\
  & \sum_{s, t} x_{i , i}^{t , s} \cdot k \leq w_{i}^{max} \cdot y_{o \left(k\right) , i} \cdot k \forall i , k \tag{eq\_0006} \\
  & \sum_{s, t} x_{i , i}^{t , s} \cdot k \geq w_{i}^{min} \cdot y_{o \left(k\right) , i} \cdot k \forall i , k \tag{eq\_0007} \\
  & \sum_{k, j, s, t \in \left(tp , tp + f\right)} x_{i , j}^{t , s} \cdot k \leq 1 \forall i , tp \tag{eq\_0008} \\
  & \sum_{k, i, t, s \in \left(tp , tp + h\right)} x_{i , j}^{t , s} \cdot k \leq 1 \forall j , tp \tag{eq\_0009} \\
  & \sum_{k, s} x_{i , i}^{t , s} \cdot k \leq u_{i} \forall i , t \tag{eq\_0010} \\
  & x_{i , j}^{tp , tp + l_{i , j} \left(k\right)} \cdot k + \sum_{s \in \left[tp , tp + l_{i , j} \left(k\right)\right], t \in \left[tp , tp + l_{i , j} \left(k\right)\right]} x_{i , j}^{t , s} kp \leq 1 \forall i , j , k , kp , tp \tag{eq\_0011} \\
