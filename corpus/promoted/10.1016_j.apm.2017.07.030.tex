  \min\quad & \left[\mathit{alpha} \cdot \sum_{k \in E} \left(\right. z_{k} + \mathit{Buff}_{k} \cdot h_{k} \left.\right) - \mathit{beta} \cdot R\right] . \tag{eq\_0001} \\

  & x_k^end - x_k^begin - Buff_k \cdot h_k \ge d_k - epsilon_k \cdot delta_k, forall \cdot k \cdot in \cdot E \tag{eq\_0004} \\
  & x_{k}^{b e g i n} \geq b_{k}^{\text{nominal}} , \forall k \in E : h_{k} = 1 \tag{eq\_0005} \\
  & x_{k}^{b e g i n} = b_{k}^{s t a t i c} , \forall k \in E : b_{k}^{s t a t i c} > 0 \tag{eq\_0006} \\
  & x_{k}^{e n d} = e_{k}^{s t a t i c} , \forall k \in E : e_{k}^{s t a t i c} > 0 \tag{eq\_0007} \\
  & x_{k}^{e n d} - \mathit{Buff}_{k} \cdot h_{k} - e_{k}^{\text{nominal}} \leq z_{k} , \forall k \in E \tag{eq\_0008} \\
  & \sum_{t c \in P_{j}} q_{k , j , t c} = 1 , \forall k \in L_{j} , j \in B \tag{eq\_0009} \\
  & q_{k , j , t c} + q_{\hat{k} , j , t c} - 1 \leq \mathit{lambda}_{k , \hat{k}} + \mathit{gamma}_{k , \hat{k}} , \forall k , \hat{k} \in L_{j} , t c \in P_{j} , j \in B ; k < \hat{k} \tag{eq\_0010} \\
  & x_{\hat{k}}^{b e g i n} - x_{k}^{e n d} \geq \mathit{Delta}_{j}^{M} \mathit{gamma}_{k , \hat{k}} - M \left(1 - \mathit{gamma}_{k , \hat{k}}\right) , \forall k , \hat{k} \in L_{j} , j \in B ; k < \hat{k} , o_{\hat{k}} \neq o_{k} \tag{eq\_0011} \\
  & x_{\hat{k}}^{b e g i n} - x_{k}^{e n d} \geq \mathit{Delta}_{j}^{F} \mathit{gamma}_{k , \hat{k}} - M \left(1 - \mathit{gamma}_{k , \hat{k}}\right) , \forall k , \hat{k} \in L_{j} , j \in B ; k < \hat{k} , o_{\hat{k}} = o_{k} \tag{eq\_0012} \\
  & x_{k}^{b e g i n} - x_{\hat{k}}^{e n d} \geq \mathit{Delta}_{j}^{M} \mathit{lambda}_{k , \hat{k}} - M \left(1 - \mathit{lambda}_{k , \hat{k}}\right) , \forall k , \hat{k} \in L_{j} , j \in B ; k < \hat{k} , o_{\hat{k}} \neq o_{k} \tag{eq\_0013} \\
  & x_{k}^{b e g i n} - x_{\hat{k}}^{e n d} \geq \mathit{Delta}_{j}^{F} \mathit{lambda}_{k , \hat{k}} - M \left(1 - \mathit{lambda}_{k , \hat{k}}\right) , \forall k , \hat{k} \in L_{j} , j \in B ; k < \hat{k} , o_{\hat{k}} = o_{k} \tag{eq\_0014} \\
  & \mathit{lambda}_{k , \hat{k}} + \mathit{gamma}_{k , \hat{k}} \leq 1 \forall k , \hat{k} \in L_{j} , j \in B ; k < \hat{k} \tag{eq\_0015} \\
  & g_{i}^{t r a i n} \cdot q_{k , j , t c} \leq g_{j , t c}^{t r a c k} \forall k \in \left(K_{i} \cap L_{j}\right) , t c \in P_{j} , j \in B , i \in T \tag{eq\_0016} \\
  & z_{k} + \mathit{Buff}_{k} \cdot h_{k} - w_{k} \leq M \mathit{varepsilon}_{k} \forall k \in E \tag{eq\_0017} \\
  & x_{\hat{k}}^{b e g i n} - x_{k}^{e n d} \geq g_{k , \hat{k}}^{c o n n e c t i o n} \forall k , \hat{k} \in E ; k , \hat{k} \in E^{c o n n e c t i o n} \tag{eq\_0018} \\
  & x_{k}^{b e g i n} , x_{k}^{e n d} , z_{k} \geq 0 \forall k \in E \tag{eq\_0019} \\
