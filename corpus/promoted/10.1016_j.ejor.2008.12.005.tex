\begin{align}
  & \mathit{atl}_{i , K_{i}} = 0 \tag{eq\_0001\_a} \\
  & \mathit{atl}_{i , k} = \underset{k + 1 \leq k^{'} \leq K_{i} \left|\right. \mathit{dt}_{i , k^{'}} - l_{i}^{J} < \mathit{dt}_{i , k}}{\sum} \left(\mathit{delta}_{i , k^{'}}\right) \forall i \in J , 1 \leq k < K_{i} . \tag{eq\_0001\_b} \\
  & \mathit{dt}_{i , 0} = 0 \tag{eq\_0002\_a} \\
  & \mathit{dt}_{i , k} = \mathit{dt}_{i , k - 1} + l_{m_{i , k}}^{M} 1 \leq k \leq K_{i} \tag{eq\_0002\_b} \\
  & \mathit{dt}_{i , K_{i} + 1} = \mathit{dt}_{i , K_{i}} + l_{i}^{J} \forall i \tag{eq\_0002\_c} \\
  & \mathit{fstage}_{i , k} = min \left\{\right. k^{'} \left|\right. k + 1 \leq k^{'} \leq K_{i} + 1 , \mathit{dt}_{i , k^{'}} \geq \mathit{dt}_{i , k} + l_{i}^{J} \left.\right\} \forall i , k = 1 , \ldots , K_{i} , \tag{eq\_0003} \\
  & \mathit{fpos}_{i , k} = \mathit{dt}_{i , k} + l_{i}^{J} - \mathit{dt}_{i , \mathit{fstage}_{i , k} - 1} \forall i , k . \tag{eq\_0004} \\
  & \mathit{apt}_{i , k} = d_{i , k} \cdot \mathit{ind}_{i , k} + a_{i , k} \cdot \mathit{ina}_{i , k - 1} , k = 1 , \ldots , K_{i} , \tag{eq\_0005} \\
  & \mathit{ina}_{i , 1} = 0 \tag{eq\_0007\_a} \\
  & \mathit{ina}_{i , k} = \left\{\begin{matrix} 1 & \text{for} \mathit{delta}_{i , k - 1} > 0 \\ 0 & \text{for} \mathit{delta}_{i , k - 1} = 0 \end{matrix}\right , k = 2 , \ldots , K_{i} . \tag{eq\_0007\_b} \\
  & e_{1} = \left(o_{i , k^{ \cdot }} , o_{i^{'} , k^{'}} , s_{o_{i , k} , o_{i^{'} , k^{'}}} - \mathit{delta}_{i , k^{ \cdot }} - p_{i , k^{ \cdot }} \left(1 - \mathit{fpos}_{i , k} / l_{m_{i , k^{ \cdot }}}^{M}\right)\right) , k^{ \cdot } = \mathit{fstage}_{i , k} \leq K_{i} , \tag{eq\_0014} \\
  & e_{1} = \left(o_{i , k^{ \cdot }} , o_{i^{'} , k^{'}} , s_{o_{i , k} , o_{i^{'} , k^{'}}} + p_{i , k^{ \cdot }} \left(\mathit{fpos}_{i , k} / l_{m_{i , k^{ \cdot }}}^{M}\right)\right) , k^{ \cdot } = \mathit{fstage}_{i , k} > K_{i} , \tag{eq\_0015} \\
  & e_{2} = \left(o_{i^{'} , k^{ \cdot }} , o_{i , k} , s_{o_{i^{'} , k^{'}} , o_{i , k} ,} - \mathit{delta}_{i^{'} , k^{ \cdot }} - p_{i^{'} , k^{ \cdot }} \left(1 - \mathit{fpos}_{i^{'} , k^{'}} / l_{m_{i^{'} , k^{ \cdot }}}^{M}\right)\right) , k^{ \cdot } = \mathit{fstage}_{i^{'} , k^{'}} \leq K_{i^{'}} , \tag{eq\_0016} \\
  & e_{2} = \left(o_{i^{'} , k^{ \cdot }} , o_{i , k} , s_{o_{i^{'} , k^{'}} , o_{i , k}} + p_{i^{'} , k^{ \cdot }} \left(\mathit{fpos}_{i^{'} , k^{'}} / l_{m_{i^{'} , k^{ \cdot }}}^{M}\right)\right) , k^{ \cdot } = \mathit{fstage}_{i^{'} , k^{'}} > K_{i^{'}} . \tag{eq\_0017} \\
  & \mathit{entry}_{i , k} ≔ \mathit{lpv}_{i , k} - p_{i , k} - \mathit{delta}_{i , k} - \mathit{atl}_{i , k} \forall i \in J , k = K_{i} , \ldots , 1 , \tag{eq\_0018} \\
  & \mathit{exit}_{i , k} ≔ \mathit{lpv}_{i , k} + \mathit{stl}_{i} \forall i \in J , k = K_{i} , \tag{eq\_0019} \\
  & \mathit{exit}_{i , k} ≔ \mathit{lpv}_{i , k^{ \cdot }} - \left(\mathit{delta}_{i , k^{ \cdot }} + p_{i , k^{ \cdot }} \left(1 - \mathit{fpos}_{i , k} / l_{m_{i , k^{ \cdot }}}^{M}\right)\right) - d_{i , k^{ \cdot }} \left(\mathit{ind}_{i , k^{ \cdot }}\right) \forall i \in J , k = K_{i} - 1 , \ldots , 1 \left|\right. k^{ \cdot } = \mathit{fstage}_{i , k} \leq K_{i} , \tag{eq\_0020} \\
  & \mathit{exit}_{i , k} ≔ \mathit{lpv}_{i , k^{ \cdot } - 1} + p_{i , k^{ \cdot } - 1} \left(\mathit{fpos}_{i , k} / l_{m_{i , k^{ \cdot } - 1}}^{M}\right) \forall i \in J , k = K_{i} - 1 , \ldots , 1 \left|\right. k^{ \cdot } = \mathit{fstage}_{i , k} = K_{i} + 1 , \tag{eq\_0021} \\
  & \mathit{delay}_{i , k} = \mathit{exit}_{i , k} - \left(\mathit{entry}_{i , k} + p_{i , k} + \mathit{delta}_{i , k} + \mathit{stl}_{i} + \mathit{atl}_{i , k} + \mathit{apt}_{i , k}\right) \forall i \in J , k = K_{i} , \ldots , 1 . \tag{eq\_0022} \\
  \mathit{LB}\quad & = \underset{j \in M}{max} \left\{\mathit{lb}_{j}\right\} \tag{eq\_0023\_a} \\
  & \mathit{lb}_{j} = \sum_{o_{i , k} \in U_{j}} \left(\bar{p}_{i , k}\right) + \underset{o_{i , k} \in U_{j}}{min} \left\{\mathit{tau}_{i , k}\right\} + \underset{o_{i , k} \in U_{j}}{min} \left\{\mathit{phi}_{i , k}\right\} , \forall j \in M \tag{eq\_0023\_b} \\
  & \bar{p}_{i , k} = p_{i , k} + \mathit{delta}_{i , k} + \mathit{apt}_{i , k} + \mathit{atl}_{i , k} + \mathit{stl}_{i} , \forall o_{i , k} \in O , \tag{eq\_0024} \\
  & \mathit{tau}_{i , 1} = \mathit{rlt}_{i} \tag{eq\_0025\_a} \\
  & \mathit{tau}_{i , k} = \mathit{tau}_{i , k - 1} + p_{i , k - 1}^{'} \left[= \sum_{k^{'} = 1 , \ldots , k - 1} \left(p_{i , k^{'}}^{'}\right)\right] , k = 2 , \ldots , K_{i} \tag{eq\_0025\_b} \\
  & \mathit{phi}_{i , K_{i}} = 0 \tag{eq\_0026\_a} \\
  & \mathit{phi}_{i , k} = \mathit{phi}_{i , k + 1} + p_{i , k + 1}^{''} \left[= \sum_{k^{'} = k + 1 , \ldots , K_{i}} \left(p_{i , k^{'}}^{''}\right)\right] , k = K_{i} - 1 , \ldots , 1 \tag{eq\_0026\_b} \\
  & p_{i , k}^{'} = p_{i , k} + \mathit{delta}_{i , k} + \mathit{apt}_{i , k} \tag{eq\_0027\_a} \\
  & p_{i , k}^{''} = p_{i , k} + \mathit{apt}_{i , k} + \mathit{atl}_{i , k} + \left(\mathit{delta}_{i , k} \text{if} \mathit{atl}_{i , k - 1} = 0\right) . \tag{eq\_0027\_b} \\
\end{align}
