\begin{align}
  & t_{\mathit{sl}}^{A} = t_{l}^{o} + \sum_{k = 2}^{q} t_{\mathit{lk} \left(k - 1\right)}^{R} + \sum_{k = 1}^{q - 1} t_{\mathit{lk}}^{\mathit{Dw}} , \tag{eq\_0003} \\
  & t_{sllp}^{w} = h \cdot T_{lp}^{H} + t_{slp}^{D} - \left(t_{\mathit{sl}}^{A} + T_{sllp}^{\mathit{Tra}}\right) . \tag{eq\_0005} \\
  & T_{\mathit{earliest}} \leq t_{l}^{o} \leq T_{\mathit{latest}} , \tag{eq\_0006} \\
  & T_{\mathit{earliest}} \leq t_{lp}^{o} \leq T_{\mathit{latest}} . \tag{eq\_0007} \\
  & T_{\text{min} \mathit{lk} \left(k - 1\right)}^{R} \leq t_{\mathit{lk} \left(k - 1\right)}^{R} \leq T_{\text{max} \mathit{lk} \left(k - 1\right)}^{R} , \tag{eq\_0008} \\
  & T_{\text{min} lp k \left(k - 1\right)}^{R} \leq t_{lp k \left(k - 1\right)}^{R} \leq T_{\text{max} lp k \left(k - 1\right)}^{R} . \tag{eq\_0009} \\
  & T_{min}^{\mathit{Dw}} \leq t_{\mathit{ls}}^{\mathit{Dw}} \leq T_{max}^{\mathit{Dw}} , \tag{eq\_0010} \\
  & T_{min}^{\mathit{Dw}} \leq t_{lp s}^{\mathit{Dw}} \leq T_{max}^{\mathit{Dw}} . \tag{eq\_0011} \\
  \min\quad & \sum_{s \in S \left(l\right) \cap S \left(lp\right), l \in L, lp \in L} t_{sllp}^{w} . \tag{eq\_0012} \\
\end{align}
