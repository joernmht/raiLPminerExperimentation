  \min\quad & \sum_{j = 1}^{N} \sum_{u = 1}^{S - 1} \sum_{v = u + 1}^{S} \underset{t \in \left[\mathit{TD}_{\bar{j}}^{u} , \mathit{TD}_{j}^{u}\right)}{\sum} \mathit{tau}_{j}^{u , v} \cdot P^{u , v} \left(t\right) \cdot \left(\mathit{TD}_{j}^{u} - t\right) \tag{eq\_0005} \\
