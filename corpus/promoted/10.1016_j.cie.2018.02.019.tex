%@ con eq_0032_f kind=linear domain=- indicator=- :: MPC passenger flow equation: αob,k = Λ[αob,k-1 + α_{k-1}(td_k - td_{k-1})] - Σ Λ^{θ+1} αl_k^θ (td_{k-θ} - td_{k-θ-1})
%@ con eq_0032_g kind=linear domain=- indicator=- :: MPC auxiliary energy equation: E_aux,k+1 = (P_ac·αob,k + P_lc·I)(td_{k+1} - Λ·td_k - T0,k)

\begin{align}
  & \mathit{td}_{i + 1}^{j} = \mathit{td}_{i}^{j} + r_{i}^{j} + s_{i + 1}^{j} , \tag{eq\_0001} \\
  & r_{i}^{j} = R_{i} + \mathit{ur}_{i}^{j} + \mathit{omega} r_{i}^{j} , \tag{eq\_0002} \\
  & s_{i + 1}^{j} = S_{0} + \mathit{up}_{i + 1}^{j} + \mathit{omega} p_{i + 1}^{j} + \mathit{lambda}_{i + 1}^{j} \cdot \left(\mathit{ta}_{i + 1}^{j} - \mathit{td}_{i + 1}^{j - 1}\right) , \tag{eq\_0003} \\
  & \mathit{ta}_{i + 1}^{j} = \mathit{td}_{i}^{j} + r_{i}^{j} . \tag{eq\_0004} \\
  & X_{i + 1}^{j} = \mathit{td}_{i + 1}^{j} - \mathit{Td}_{i + 1}^{j} , \tag{eq\_0005} \\
  & Y_{i + 1}^{j} = \left(\mathit{td}_{i + 1}^{j} - \mathit{td}_{i + 1}^{j - 1}\right) - H = X_{i + 1}^{j} - X_{i + 1}^{j - 1} , \tag{eq\_0006} \\
  & \mathit{ta}_{k} = \mathit{Lambda} \cdot \mathit{td}_{k} + T_{0 , k} + R + \mathit{ur}_{k} + \mathit{omega} r_{k} , \tag{eq\_0007} \\
  & \mathit{td}_{k + 1} = \mathit{Lambda} \cdot \mathit{td}_{k} + T_{0 , k} + R + \mathit{ur}_{k} + \mathit{omega} r_{k} + S_{0} + \mathit{up}_{k} + \mathit{omega} p_{k} + \mathit{lambda}_{k} \cdot \left(\mathit{ta}_{k} - \mathit{td}_{k}\right) , \tag{eq\_0008} \\
  & X_{k + 1} = \mathit{td}_{k + 1} - \mathit{Td}_{k + 1} , \tag{eq\_0009} \\
  & Y_{k + 1} = X_{k + 1} - X_{k} , \tag{eq\_0010} \\
  & \mathit{alpha}_{\mathit{ob} , i + 1}^{j} = \mathit{alpha}_{\mathit{ob} , i}^{j} + \mathit{alpha}_{i}^{j} \cdot \left(\mathit{td}_{i}^{j} - \mathit{td}_{i}^{j - 1}\right) - \sum_{l = 1}^{i - 1} \mathit{alpha} l_{l , i}^{j} \cdot \left(\mathit{td}_{l}^{j} - \mathit{td}_{l}^{j - 1}\right) , \tag{eq\_0011} \\
  & \mathit{alpha}_{\mathit{ob} , 1}^{j} = 0 , \tag{eq\_0012} \\
  & \mathit{alpha}_{\mathit{ob} , k} = \mathit{Lambda} \cdot \left[\mathit{alpha}_{\mathit{ob} , k - 1} + \mathit{alpha}_{k - 1} \cdot \left(\mathit{td}_{k} - \mathit{td}_{k - 1}\right)\right] - \sum_{\mathit{theta} = 1}^{M - 2} \mathit{Lambda}^{\mathit{theta} + 1} \left[\mathit{alpha} l_{k}^{\mathit{theta}} \cdot \left(\mathit{td}_{k - \mathit{theta}} - \mathit{td}_{k - \mathit{theta} - 1}\right)\right] , \tag{eq\_0013} \\
  & E_{i + 1}^{j} = E_{\mathit{tr} , i + 1}^{j} + E_{\mathit{aux} , i + 1}^{j} , \tag{eq\_0014} \\
  & E_{\mathit{aux} , i + 1}^{j} = \left(P_{\mathit{ac}} \cdot \mathit{alpha}_{\mathit{ob} , i + 1}^{j} + P_{\mathit{lc}}\right) \left(\mathit{td}_{i + 1}^{j} - \mathit{td}_{i}^{j}\right) , \tag{eq\_0015} \\
  & E_{\mathit{tr} 0 , i + 1}^{j} = \left(\frac{u_{1 , i}}{u_{2 , i} - r_{i}^{j}} + u_{3 , i}\right) , \tag{eq\_0016} \\
  & E_{\mathit{tr} , i + 1}^{j} = \left(1 + \mathit{omega}_{p} \mathit{alpha}_{\mathit{ob} , i + 1}^{j}\right) \left(\frac{u_{1 , i}}{u_{2 , i} - r_{i}^{j}} + u_{3 , i}\right) , \tag{eq\_0017} \\
  & E_{k + 1} = E_{\mathit{tr} , k + 1} + E_{\mathit{aux} , k + 1} , \tag{eq\_0018} \\
  & E_{\mathit{aux} , k + 1} = \left(P_{\mathit{ac}} \cdot \mathit{alpha}_{\mathit{ob} , k} + P_{\mathit{lc}} \cdot I\right) \left(\mathit{td}_{k + 1} - \left(\mathit{Lambda} \cdot \mathit{td}_{k} + T_{0 , k}\right)\right) , \tag{eq\_0019} \\
  & E_{\mathit{tr} , k + 1} = \left(I + \mathit{omega}_{p} \mathit{alpha}_{\mathit{ob} , k}^{'}\right) \cdot E_{\mathit{tr} 0 , k + 1} , \tag{eq\_0020} \\
  & J = p \sum_{i , j} \left(X_{i + 1}^{j}\right)^{2} + q \sum_{i , j} \left(Y_{i + 1}^{j}\right)^{2} + a \sum_{i , j} \left(\mathit{ur}_{i}^{j}\right)^{2} + b \sum_{i , j} \left(\mathit{up}_{i}^{j}\right)^{2} + c \sum_{i , j} E_{i + 1}^{j} , \tag{eq\_0021} \\
  & J_{k} = p \cdot X_{k + 1}^{T} \cdot X_{k + 1} + q \cdot Y_{k + 1}^{T} \cdot Y_{k + 1} + a \cdot \mathit{ur}_{k}^{T} \cdot \mathit{ur}_{k} + b \cdot \mathit{up}_{k}^{T} \cdot \mathit{up}_{k} + C \cdot E_{k + 1} , \tag{eq\_0022} \\
  & \mathit{ta}_{i + 1}^{j} - \mathit{td}_{i + 1}^{j - 1} \geq I_{i + 1 , min} , \tag{eq\_0023} \\
  & s_{i + 1}^{j} \geq S_{0} , \tag{eq\_0024} \\
  & \mathit{up}_{i + 1}^{j} + \mathit{omega} p_{i + 1}^{j} + \mathit{lambda}_{i + 1}^{j} \cdot \left(\mathit{ta}_{i + 1}^{j} - \mathit{td}_{i + 1}^{j - 1}\right) \geq 0 . \tag{eq\_0025} \\
  & \mathit{UR}_{i , min} \leq \mathit{ur}_{i}^{j} \leq \mathit{UR}_{i , max} , \tag{eq\_0026} \\
  & \mathit{UP}_{i , min} \leq \mathit{up}_{i}^{j} \leq \mathit{UP}_{i , max} , \tag{eq\_0027} \\
  & \mathit{ta}_{k} - \mathit{td}_{k} \geq I_{min} , \tag{eq\_0028} \\
  & \mathit{up}_{k} + \mathit{omega} p_{k} + \mathit{lambda}_{k} \left(\mathit{ta}_{k} - \mathit{td}_{k}\right) \geq 0 , \tag{eq\_0029} \\
  & \mathit{UR}_{\mathit{min}} \leq \mathit{ur}_{k} \leq \mathit{UR}_{\mathit{max}} , \tag{eq\_0030} \\
  & \mathit{UP}_{\mathit{min}} \leq \mathit{up}_{k} \leq \mathit{UP}_{\mathit{max}} , \tag{eq\_0031} \\
  \min\quad & \sum_{j_{0}}^{j_{f}} p \cdot X_{k + 1}^{T} \cdot X_{k + 1} + q \cdot Y_{k + 1}^{T} \cdot Y_{k + 1} + a \cdot \mathit{ur}_{k}^{T} \cdot \mathit{ur}_{k} + b \cdot \mathit{up}_{k}^{T} \cdot \mathit{up}_{k} + C \cdot E_{k + 1}^{T} \tag{eq\_0032\_a} \\
  & \mathit{ta}_{k} = \mathit{Lambda} \cdot \mathit{td}_{k} + T_{0 , k} + R + \mathit{ur}_{k} + \mathit{omega} r_{k} \tag{eq\_0032\_b} \\
  & \mathit{td}_{k + 1} = \mathit{Lambda} \cdot \mathit{td}_{k} + T_{0 , k} + R + \mathit{ur}_{k} + \mathit{omega} r_{k} + S_{0} + \mathit{up}_{k} + \mathit{omega} p_{k} + \mathit{lambda}_{k} \cdot \left(\mathit{ta}_{k} - \mathit{td}_{k}\right) \tag{eq\_0032\_c} \\
  & X_{k + 1} = t_{k + 1} - T_{k + 1} \tag{eq\_0032\_d} \\
  & Y_{k + 1} = X_{k + 1} - X_{k} \tag{eq\_0032\_e} \\
  & \mathit{alpha}_{\mathit{ob} , k} = \mathit{Lambda} \cdot \left[\mathit{alpha}_{\mathit{ob} , k - 1} + \mathit{alpha}_{k - 1} \cdot \left(\mathit{td}_{k} - \mathit{td}_{k - 1}\right)\right] - \sum_{\mathit{theta} = 1}^{M - 2} \mathit{Lambda}^{\mathit{theta} + 1} \left[\mathit{alpha} l_{k}^{\mathit{theta}} \cdot \left(\mathit{td}_{k - \mathit{theta}} - \mathit{td}_{k - \mathit{theta} - 1}\right)\right] \tag{eq\_0032\_f} \\
  & E_{\mathit{aux} , k + 1} = \left(P_{\mathit{ac}} \cdot \mathit{alpha}_{\mathit{ob} , k} + P_{\mathit{lc}} \cdot I\right) \cdot \left(\mathit{td}_{k + 1} - \left(\mathit{Lambda} \cdot \mathit{td}_{k} + T_{0 , k}\right)\right) \tag{eq\_0032\_g} \\
  & E_{\mathit{tr} , k + 1} = \left(I + \mathit{omega}_{p} \mathit{alpha}_{\mathit{ob} , k}\right) E_{\mathit{tr} 0 , k + 1} \tag{eq\_0032\_h} \\
  & E_{k + 1} = E_{\mathit{tr} , k + 1} + E_{\mathit{aux} , k + 1} \tag{eq\_0032\_i} \\
  & \mathit{ta}_{k} - \mathit{td}_{k} \geq I_{min} , k = j_{0} , \ldots , j_{f} \tag{eq\_0032\_j} \\
  & \mathit{up}_{k} + \mathit{omega} p_{k} + \mathit{lambda}_{k} \left(\mathit{ta}_{k} - \mathit{td}_{k}\right) \geq 0 , k = j_{0} , \ldots , j_{f} \tag{eq\_0032\_k} \\
  & \mathit{UR}_{min} \leq \mathit{ur}_{k} \leq \mathit{UR}_{max} , k = j_{0} , \ldots , j_{f} \tag{eq\_0032\_l} \\
  & \mathit{UP}_{min} \leq \mathit{up}_{k} \leq \mathit{UP}_{max} , k = j_{0} , \ldots , j_{f} \tag{eq\_0032\_m} \\
\end{align}
