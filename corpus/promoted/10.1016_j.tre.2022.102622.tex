  & t_{p} = \sum_{s \in \mathcal{S}_{p}} f_{s} + t_{t r a i n} , \tag{eq\_0001} \\

  & \mathit{gamma}_{p , q}^{1} = \mathit{sigma}_{p}^{l} + t_{p} + \mathit{rho} , \tag{eq\_0005} \\
  & \mathit{gamma}_{p , q}^{2} = \mathit{sigma}_{p}^{l} + \sum_{i = 1}^{4} f_{s_{i}} + \mathit{rho} < \mathit{gamma}_{p , q}^{1} . \tag{eq\_0006} \\
  & \mathit{gamma}_{p , q}^{3} = \mathit{sigma}_{p}^{l} + \sum_{i = 1}^{4} f_{s_{i}} + \mathit{rho} - \left(f_{s_{6}} + \mathit{sigma}_{q}^{c} + \mathit{sigma}_{q}^{l}\right) < \mathit{gamma}_{p , q}^{2} , \tag{eq\_0007} \\
  & \mathit{gamma}_{p , q}^{2} = \mathit{sigma}_{p}^{l} + t_{p} + \mathit{rho} - \sum_{s \in \mathcal{S}_{p , q}^{p +}} f_{s} ; \tag{eq\_0010} \\
  & \mathit{gamma}_{p , q}^{3} = \mathit{sigma}_{p}^{l} + t_{p} + \mathit{rho} - \sum_{s \in \mathcal{S}_{p , q}^{p +}} f_{s} - \left(\sum_{s \in \mathcal{S}_{p , q}^{q -}} f_{s} + \mathit{sigma}_{q}^{l} + \mathit{sigma}_{q}^{c}\right) . \tag{eq\_0011} \\
  & \mathit{tau}_{kp , q} \geq \mathit{tau}_{k , p} + \mathit{gamma}_{p , q}^{\mathit{pi}} . \tag{eq\_0012} \\
  & y_{k}^{d} - y_{k}^{a} \geq \mathit{Omega}_{k} , \forall k \in \mathcal{K} , \tag{eq\_0013} \\
  & y_{k}^{a} \geq t_{k}^{a} , \forall k \in \mathcal{K} , \tag{eq\_0014} \\
  & y_{k}^{d} \geq t_{k}^{d} , \forall k \in \mathcal{K} , \tag{eq\_0016} \\
  & o_{k , u} = \sum_{p \in \mathcal{R}^{r}} \left(x_{k , p} \cdot \mathit{alpha}_{p , u}\right) , \forall k \in \mathcal{K} , \forall u \in \mathcal{I} , \tag{eq\_0017} \\
  & d_{k , v} = \sum_{p \in \mathcal{R}^{d}} \left(x_{k , p} \cdot \mathit{beta}_{p , v}\right) , \forall k \in \mathcal{K} , \forall v \in \mathcal{O} . \tag{eq\_0019} \\
  & w_{k , i} = \sum_{p \in \mathcal{R}^{r}} \left(x_{k , p} \cdot \mathit{theta}_{p , i}\right) , \forall k \in \mathcal{K} , i \in \mathcal{T} , \tag{eq\_0020} \\
  & w_{k , i} = \sum_{p \in \mathcal{R}^{d}} \left(x_{k , p} \cdot \mathit{delta}_{p , i}\right) , \forall k \in \mathcal{K} , i \in \mathcal{T} . \tag{eq\_0022} \\
  & \sum_{i \in \mathcal{T}} w_{k , i} = 1 , \forall k \in \mathcal{K} . \tag{eq\_0023} \\
  & M \cdot \left(2 + B_{k , kp} - w_{k , i} - w_{kp , i}\right) + y_{kp}^{r} - y_{k}^{d} \geq \mathit{rho} , \forall k , kp \in \mathcal{K} , k < kp , \forall i \in \mathcal{T} , \tag{eq\_0024} \\

  & \mathit{tau}_{kp , pp} - \mathit{tau}_{k , p} + M \cdot \left(2 + V_{k , kp}^{\mathit{phi}} - x_{k , p} - x_{kp , pp}\right) \geq \mathit{gamma}_{p , pp}^{\mathit{pi}} , \forall k , kp \in \mathcal{K} , k \neq kp , \forall p , pp \text{such}\textrm{ }\text{that} \mathit{gamma}_{p , pp}^{\mathit{pi}} > 0 , \tag{eq\_0029} \\
